   792→  for (let i = 0; i < cap; i++) {
   793→    const c = getCostForLevel(u, u.level + i);
   794→    if (total + c > money) break;
   795→    total += c; count++;
   796→  }
   797→  return count;
   798→};

   829→function recalcStats(upgrades: Upgrade[]) {
   830→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   831→  let goldenChance = 0.03; let critChance = 0.05;
   832→  for (const u of upgrades) {
   833→    for (let i = 0; i < u.level; i++) {
   834→      switch (u.effect) {
   835→        case 'clickPower': clickPower += u.value; break;
   836→        case 'autoRate': autoRate += u.value; break;
   837→        case 'multiplier': multiplier += u.value; break;
   838→        case 'goldenChance': goldenChance += u.value; break;
   839→        case 'critChance': critChance += u.value; break;
   840→      }
   841→    }
   842→  }
   843→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   844→}
