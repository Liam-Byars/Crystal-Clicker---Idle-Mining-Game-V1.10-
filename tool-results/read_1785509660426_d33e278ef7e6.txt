     4→// ====== Interfaces ======
     5→export interface Upgrade {
     6→  id: string;
     7→  name: string;
     8→  description: string;
     9→  icon: string;
    10→  baseCost: number;
    11→  costMultiplier: number;
    12→  level: number;
    13→  effect: 'clickPower' | 'autoRate' | 'multiplier' | 'goldenChance' | 'critChance';
    14→  value: number;
    15→  maxLevel?: number;
    16→}
    17→
    18→export interface Achievement {
    19→  id: string;
    20→  name: string;
    21→  description: string;
    22→  icon: string;
    23→  unlocked: boolean;
    24→  unlockedAt?: number;
    25→  condition: (state: GameState) => boolean;
    26→}
    27→
    28→export interface FloatingText {
    29→  id: number;
    30→  value: number;
    31→  x: number;
    32→  y: number;
    33→  type: 'normal' | 'golden' | 'combo' | 'crit' | 'powerup' | 'event' | 'offline' | 'milestone';
    34→}
    35→
    36→export interface PowerUp {
    37→  id: string;
    38→  name: string;
    39→  icon: string;
    40→  effect: 'doubleClick' | 'tripleAuto' | 'instantBonus';
    41→  duration: number;
    42→  timer: number;
    43→  value: number;
    44→}
    45→
    46→export interface GameEvent {
    47→  id: string;
    48→  name: string;
    49→  description: string;
    50→  icon: string;
    51→  effect: 'doubleAuto' | 'doubleClick' | 'tripleGolden';
    52→  duration: number;
    53→  timer: number;
    54→  value: number;
    55→}
    56→
    57→export interface Milestone {
    58→  id: number;
    59→  value: number;
    60→  label: string;
    61→  icon: string;
    62→  celebrated: boolean;
    63→}
    64→
    65→export interface Area {
    66→  id: string;
    67→  name: string;
    68→  location: string;
    69→  flag: string;
    70→  gem: string;
    71→  description: string;
    72→  icon: string;
    73→  unlockAt: number;
    74→  gradient: string;
    75→  glowColor: string;
    76→  bgAccent: string;
    77→}
    78→
    79→export type BuyQuantity = 1 | 10 | 100 | 'max';
    80→
    81→export interface GameState {
    82→  // Core resources
    83→  crystals: number;
    84→  totalClicks: number;
    85→  totalEarned: number;
    86→  clickPower: number;
    87→  multiplier: number;
    88→  autoRate: number;
    89→
    90→  // Prestige
    91→  prestige: number;
    92→  prestigePoints: number;
    93→
    94→  // Combo
    95→  combo: number;
    96→  comboTimer: number;
    97→  maxCombo: number;
    98→  lastClickTime: number;
    99→
   100→  // Click speed tracking
   101→  clickTimestamps: number[];
   102→  clicksPerSecond: number;
   103→  bestSessionCps: number;
   104→
   105→  // Critical hits
   106→  critChance: number;
   107→  critMultiplier: number;
   108→  totalCrits: number;
   109→
   110→  // Golden Crystal
   111→  goldenClicks: number;
   112→  goldenChance: number;
   113→  goldenActive: boolean;
   114→  goldenTimer: number;
   115→  goldenClickValue: number;
   116→
   117→  // Power-ups
   118→  activePowerUp: PowerUp | null;
   119→  powerUpTimer: number;
   120→
   121→  // Game Events
   122→  activeEvent: GameEvent | null;
   123→  eventTimer: number;
   124→  totalEvents: number;
   125→
   126→  // Buy quantity
   127→  buyQuantity: BuyQuantity;
   128→
   129→  // Session tracking
   130→  sessionStartTime: number;
   131→  sessionClicks: number;
   132→  sessionEarned: number;
   133→
   134→  // Milestones
   135→  milestones: Milestone[];
   136→
   137→  // Offline earnings
   138→  lastOnlineTime: number;
   139→  offlineEarned: number;
   140→  showOfflineBonus: boolean;
   141→
   142→  // Floating texts
   143→  floatingTexts: FloatingText[];
   144→  floatingTextId: number;
   145→
   146→  // Achievement notifications
   147→  achievementQueue: Achievement[];
   148→  currentNotification: Achievement | null;
   149→  notificationTimer: number;
   150→
   151→  // Click ripples
   152→  ripples: { id: number; x: number; y: number; type: FloatingText['type'] }[];
   153→  rippleId: number;
   154→
   155→  // Upgrades & Achievements
   156→  upgrades: Upgrade[];
   157→  achievements: Achievement[];
   158→
   159→  // Area system
   160→  currentArea: string;
   161→  unlockedAreas: string[];
   162→
   163→  // UI state
   164→  screenShake: boolean;
   165→  crystalPulse: number;
   166→  activeTab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'map';
   167→
   168→  // Actions
   169→  click: (x: number, y: number) => void;
   170→  clickGolden: (x: number, y: number) => void;
   171→  buyUpgrade: (upgradeId: string) => boolean;
   172→  setBuyQuantity: (q: BuyQuantity) => void;
   173→  performPrestige: () => void;
   174→  updateCombo: () => void;
   175→  updateGolden: () => void;
   176→  updatePowerUp: () => void;
   177→  updateEvent: () => void;
   178→  updateNotification: () => void;
   179→  updateClickSpeed: () => void;
   180→  checkMilestones: () => void;
   181→  addFloatingText: (x: number, y: number, value: number, type?: FloatingText['type']) => void;
   182→  removeFloatingText: (id: number) => void;
   183→  setActiveTab: (tab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'map') => void;
   184→  switchArea: (areaId: string) => void;
   185→  checkAreaUnlocks: () => void;
   186→  checkAchievements: () => void;
   187→  resetGame: () => void;
   188→  loadSave: (data: Record<string, unknown>) => void;
   189→  getSaveData: () => Record<string, unknown>;
   190→  claimOfflineEarnings: () => void;
   191→  dismissOfflineBonus: () => void;
   192→  claimReward: (amount: number, prestige?: number) => void;
   193→}

   806→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   807→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;
   808→  let total = 0; let count = 0;
   809→  for (let i = 0; i < cap; i++) {
   810→    const c = getCostForLevel(u, u.level + i);
   811→    if (total + c > money) break;
   812→    total += c; count++;
   813→  }
   814→  return count;
   815→};

   846→function recalcStats(upgrades: Upgrade[]) {
   847→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   848→  let goldenChance = 0.03; let critChance = 0.05;
   849→  for (const u of upgrades) {
   850→    for (let i = 0; i < u.level; i++) {
   851→      switch (u.effect) {
   852→        case 'clickPower': clickPower += u.value; break;
   853→        case 'autoRate': autoRate += u.value; break;
   854→        case 'multiplier': multiplier += u.value; break;
   855→        case 'goldenChance': goldenChance += u.value; break;
   856→        case 'critChance': critChance += u.value; break;
   857→      }
   858→    }
   859→  }
   860→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   861→}
