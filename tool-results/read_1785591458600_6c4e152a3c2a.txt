    30→function toLogSafe(n: number): number {
    31→  if (!isFinite(n) || n <= 0) return -Infinity;
    32→  return Math.log10(n);
    33→}
    34→function fmtExpLog(log: number): string {
    35→  if (log < 0) return '0';
    36→  if (log < 3) {
    37→    const v = Math.pow(10, log);
    38→    return v < 10 ? v.toFixed(1) : Math.floor(v).toString();
    39→  }
    40→  if (log < 6) return (Math.pow(10, log - 3)).toFixed(1) + 'K';
    41→  if (log < 9) return (Math.pow(10, log - 6)).toFixed(2) + 'M';
    42→  if (log < 12) return (Math.pow(10, log - 9)).toFixed(2) + 'B';
    43→  if (log < 15) return (Math.pow(10, log - 12)).toFixed(2) + 'T';
    44→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    45→  const tier = Math.floor((log - 15) / 3);
    46→  if (tier >= 0 && tier < 676) {
    47→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    48→    const second = String.fromCharCode(65 + (tier % 26));
    49→    const divisor_log = 15 + tier * 3;
    50→    return (Math.pow(10, log - divisor_log)).toFixed(2) + first + second;
    51→  }
    52→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ)
    53→  const tier2 = tier - 676;
    54→  if (tier2 >= 0 && tier2 < 17576) {
    55→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    56→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    57→    const c = String.fromCharCode(65 + (tier2 % 26));
    58→    const divisor_log = 15 + tier * 3;
    59→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c;
    60→  }
    61→  // Beyond ZZZ: quad-letter suffixes
    62→  const tier3 = tier2 - 17576;
    63→  if (tier3 >= 0 && tier3 < 456976) {
    64→    const a = String.fromCharCode(65 + Math.floor(tier3 / 17576));
    65→    const b = String.fromCharCode(65 + Math.floor((tier3 % 17576) / 676));
    66→    const c = String.fromCharCode(65 + Math.floor(((tier3 % 17576) % 676) / 26));
    67→    const d = String.fromCharCode(65 + (tier3 % 26));
    68→    const divisor_log = 15 + tier * 3;
    69→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c + d;
    70→  }
    71→  return Math.pow(10, log).toExponential(2);
    72→}

    74→function fmt(n: number, exp = 0): string {
    75→  if (n < 0) return '-' + fmt(-n, exp);
    76→  if (!isFinite(n)) return fmtExpLog(exp + 400);
    77→  if (exp > 0) return fmtExpLog(Math.log10(n) + exp);
    78→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    79→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    80→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    81→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    82→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    83→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    84→  const e = Math.floor(Math.log10(n));
    85→  const tier = Math.floor((e - 15) / 3);
    86→  if (tier >= 0 && tier < 676) {
    87→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    88→    const second = String.fromCharCode(65 + (tier % 26));
    89→    const divisor = Math.pow(10, 15 + tier * 3);
    90→    return (n / divisor).toFixed(2) + first + second;
    91→  }
    92→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ, 17,576 more tiers)
    93→  const tier2 = tier - 676;
    94→  if (tier2 >= 0 && tier2 < 17576) {
    95→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    96→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    97→    const c = String.fromCharCode(65 + (tier2 % 26));
    98→    const divisor = Math.pow(10, 15 + tier * 3);
    99→    return (n / divisor).toFixed(2) + a + b + c;
   100→  }
   101→  // Truly beyond all suffixes
   102→  return n.toExponential(2);
   103→}

   105→function fmtTime(ms: number): string {
   106→  const s = Math.floor(ms / 1000);
   107→  const h = Math.floor(s / 3600);
   108→  const m = Math.floor((s % 3600) / 60);
   109→  const sec = s % 60;
   110→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
   111→  return `${m}:${String(sec).padStart(2, '0')}`;
   112→}

   114→function fmtTimer(ticks: number): string {
   115→  const sec = Math.ceil(ticks / 10);
   116→  const m = Math.floor(sec / 60);
   117→  const s = sec % 60;
   118→  return `${m}:${String(s).padStart(2, '0')}`;
   119→}

   127→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
   128→  if (!soundOn || !audioCtx) return;
   129→  try {
   130→    const o = audioCtx.createOscillator();
   131→    const g = audioCtx.createGain();
   132→    o.type = type;
   133→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
   134→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
   135→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   136→    o.connect(g);
   137→    g.connect(audioCtx.destination);
   138→    o.start();
   139→    o.stop(audioCtx.currentTime + dur);
   140→  } catch { /* ignore audio errors */ }
   141→}
