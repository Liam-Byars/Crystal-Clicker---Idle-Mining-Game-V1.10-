    29→    29→function fmt(n: number): string {
    30→    30→  if (n < 0) return '-' + fmt(-n);
    31→    31→  if (!isFinite(n)) {
    32→    32→    // Number overflowed JS max — show as ∞ZZ+
    33→    33→    return '∞ZZ+';
    34→    34→  }
    35→    35→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    36→    36→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    37→    37→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    38→    38→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    39→    39→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    40→    40→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    41→    41→  const exp = Math.floor(Math.log10(n));
    42→    42→  const tier = Math.floor((exp - 15) / 3);
    43→    43→  if (tier >= 0 && tier < 676) {
    44→    44→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    45→    45→    const second = String.fromCharCode(65 + (tier % 26));
    46→    46→    const divisor = Math.pow(10, 15 + tier * 3);
    47→    47→    return (n / divisor).toFixed(2) + first + second;
    48→    48→  }
    49→    49→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ, 17,576 more tiers)
    50→    50→  const tier2 = tier - 676;
    51→    51→  if (tier2 >= 0 && tier2 < 17576) {
    52→    52→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    53→    53→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    54→    54→    const c = String.fromCharCode(65 + (tier2 % 26));
    55→    55→    const divisor = Math.pow(10, 15 + tier * 3);
    56→    56→    return (n / divisor).toFixed(2) + a + b + c;
    57→    57→  }
    58→    58→  // Truly beyond all suffixes
    59→    59→  return n.toExponential(2);
    60→    60→}

    62→    62→function fmtTime(ms: number): string {
    63→    63→  const s = Math.floor(ms / 1000);
    64→    64→  const h = Math.floor(s / 3600);
    65→    65→  const m = Math.floor((s % 3600) / 60);
    66→    66→  const sec = s % 60;
    67→    67→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    68→    68→  return `${m}:${String(sec).padStart(2, '0')}`;
    69→    69→}

    84→    84→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
    85→    85→  if (!soundOn || !audioCtx) return;
    86→    86→  try {
    87→    87→    const o = audioCtx.createOscillator();
    88→    88→    const g = audioCtx.createGain();
    89→    89→    o.type = type;
    90→    90→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    91→    91→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    92→    92→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    93→    93→    o.connect(g);
    94→    94→    g.connect(audioCtx.destination);
    95→    95→    o.start();
    96→    96→    o.stop(audioCtx.currentTime + dur);
    97→    97→  } catch { /* ignore audio errors */ }
    98→    98→}
