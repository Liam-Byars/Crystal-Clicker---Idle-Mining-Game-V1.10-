    31→function toLogSafe(n: number): number {
    32→  if (!isFinite(n) || n <= 0) return -Infinity;
    33→  return Math.log10(n);
    34→}
    35→function fmtExpLog(log: number): string {
    36→  if (log < 0) return '0';
    37→  if (log < 3) {
    38→    const v = Math.pow(10, log);
    39→    return v < 10 ? v.toFixed(1) : Math.floor(v).toString();
    40→  }
    41→  if (log < 6) return (Math.pow(10, log - 3)).toFixed(1) + 'K';
    42→  if (log < 9) return (Math.pow(10, log - 6)).toFixed(2) + 'M';
    43→  if (log < 12) return (Math.pow(10, log - 9)).toFixed(2) + 'B';
    44→  if (log < 15) return (Math.pow(10, log - 12)).toFixed(2) + 'T';
    45→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    46→  const tier = Math.floor((log - 15) / 3);
    47→  if (tier >= 0 && tier < 676) {
    48→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    49→    const second = String.fromCharCode(65 + (tier % 26));
    50→    const divisor_log = 15 + tier * 3;
    51→    return (Math.pow(10, log - divisor_log)).toFixed(2) + first + second;
    52→  }
    53→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ)
    54→  const tier2 = tier - 676;
    55→  if (tier2 >= 0 && tier2 < 17576) {
    56→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    57→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    58→    const c = String.fromCharCode(65 + (tier2 % 26));
    59→    const divisor_log = 15 + tier * 3;
    60→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c;
    61→  }
    62→  // Beyond ZZZ: quad-letter suffixes
    63→  const tier3 = tier2 - 17576;
    64→  if (tier3 >= 0 && tier3 < 456976) {
    65→    const a = String.fromCharCode(65 + Math.floor(tier3 / 17576));
    66→    const b = String.fromCharCode(65 + Math.floor((tier3 % 17576) / 676));
    67→    const c = String.fromCharCode(65 + Math.floor(((tier3 % 17576) % 676) / 26));
    68→    const d = String.fromCharCode(65 + (tier3 % 26));
    69→    const divisor_log = 15 + tier * 3;
    70→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c + d;
    71→  }
    72→  return Math.pow(10, log).toExponential(2);
    73→}

    75→function fmt(n: number, exp = 0): string {
    76→  if (n < 0) return '-' + fmt(-n, exp);
    77→  if (!isFinite(n)) return fmtExpLog(exp + 400);
    78→  if (exp > 0) return fmtExpLog(Math.log10(n) + exp);
    79→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    80→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    81→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    82→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    83→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    84→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    85→  const e = Math.floor(Math.log10(n));
    86→  const tier = Math.floor((e - 15) / 3);
    87→  if (tier >= 0 && tier < 676) {
    88→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    89→    const second = String.fromCharCode(65 + (tier % 26));
    90→    const divisor = Math.pow(10, 15 + tier * 3);
    91→    return (n / divisor).toFixed(2) + first + second;
    92→  }
    93→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ, 17,576 more tiers)
    94→  const tier2 = tier - 676;
    95→  if (tier2 >= 0 && tier2 < 17576) {
    96→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    97→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    98→    const c = String.fromCharCode(65 + (tier2 % 26));
    99→    const divisor = Math.pow(10, 15 + tier * 3);
   100→    return (n / divisor).toFixed(2) + a + b + c;
   101→  }
   102→  // Truly beyond all suffixes
   103→  return n.toExponential(2);
   104→}

   106→function fmtTime(ms: number): string {
   107→  const s = Math.floor(ms / 1000);
   108→  const h = Math.floor(s / 3600);
   109→  const m = Math.floor((s % 3600) / 60);
   110→  const sec = s % 60;
   111→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
   112→  return `${m}:${String(sec).padStart(2, '0')}`;
   113→}

   115→function fmtTimer(ticks: number): string {
   116→  const sec = Math.ceil(ticks / 10);
   117→  const m = Math.floor(sec / 60);
   118→  const s = sec % 60;
   119→  return `${m}:${String(s).padStart(2, '0')}`;
   120→}

   128→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
   129→  if (!soundOn || !audioCtx) return;
   130→  try {
   131→    const o = audioCtx.createOscillator();
   132→    const g = audioCtx.createGain();
   133→    o.type = type;
   134→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
   135→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
   136→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   137→    o.connect(g);
   138→    g.connect(audioCtx.destination);
   139→    o.start();
   140→    o.stop(audioCtx.currentTime + dur);
   141→  } catch { /* ignore audio errors */ }
   142→}
