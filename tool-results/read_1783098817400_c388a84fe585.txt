   489→   489→function recalcStats(areas: Area[]): { clickPower: number; autoRate: number; multiplier: number; goldenChance: number; critChance: number } {
   490→   490→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   491→   491→  let goldenChance = 0.03; let critChance = 0.05;
   492→   492→  for (const area of areas) {
   493→   493→    if (!area.unlocked) continue;
   494→   494→    for (const u of area.upgrades) {
   495→   495→      for (let i = 0; i < u.level; i++) {
   496→   496→        switch (u.effect) {
   497→   497→          case 'clickPower': clickPower += u.value; break;
   498→   498→          case 'autoRate': autoRate += u.value; break;
   499→   499→          case 'multiplier': multiplier += u.value; break;
   500→   500→          case 'goldenChance': goldenChance += u.value; break;
   501→   501→          case 'critChance': critChance += u.value; break;
   502→   502→        }
   503→   503→      }
   504→   504→    }
   505→   505→  }
   506→   506→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.8), critChance: Math.min(critChance, 0.95) };
   507→   507→}

   690→   690→        const evBonus = s.activeEvent?.effect === 'doubleAuto' ? s.activeEvent.value : 1;
   691→   691→        rate *= puBonus * evBonus;

   693→   693→        const income = rate * prestMult * 0.1;
   694→   694→        updates.crystals = Math.round((s.crystals + income) * 100) / 100;
   695→   695→        updates.totalEarned = Math.round((s.totalEarned + income) * 100) / 100;
   696→   696→        updates.sessionEarned = Math.round((s.sessionEarned + income) * 100) / 100;
   697→   697→      }
   698→   698→
   699→   699→      // Combo decay
   700→   700→      if (s.comboTimer > 0) {
   701→   701→        updates.comboTimer = s.comboTimer - 1;
   702→   702→        if (s.comboTimer <= 1) updates.combo = 0;
   703→   703→      }
   704→   704→
   705→   705→      // Golden crystal
   706→   706→      if (s.goldenActive) {
   707→   707→        if (s.goldenTimer > 0) updates.goldenTimer = s.goldenTimer - 1;
   708→   708→        else { updates.goldenActive = false; updates.goldenTimer = 0; }
   709→   709→      } else {
   710→   710→        let chance = 0.001;
   711→   711→        if (s.activeEvent?.effect === 'tripleGolden') chance *= s.activeEvent.value;
   712→   712→        chance *= (s.goldenChance / 0.03);
   713→   713→        if (Math.random() < chance) {
   714→   714→          const bv = s.clickPower * s.multiplier * 10 * (1 + s.prestigePoints * 0.1);
   715→   715→          updates.goldenActive = true;
   716→   716→          updates.goldenTimer = 400;
   717→   717→          updates.goldenClickValue = Math.round(bv * 10) / 10;
   718→   718→        }
   719→   719→      }
   720→   720→
   721→   721→      // Event timer
   722→   722→      if (s.activeEvent) {
   723→   723→        if (s.eventTimer > 0) updates.eventTimer = s.eventTimer - 1;
   724→   724→        else { updates.activeEvent = null; updates.eventTimer = 0; }
   725→   725→      } else if (s.totalClicks > 50 && Math.random() < 0.0003) {
   726→   726→        const t = EVENT_DEFS[Math.floor(Math.random() * EVENT_DEFS.length)];
   727→   727→        updates.activeEvent = { ...t, timer: t.duration };
   728→   728→        updates.eventTimer = t.duration;
   729→   729→        updates.totalEvents = s.totalEvents + 1;
   730→   730→        const q = [...s.achievementQueue];
   731→   731→        q.push({ id: `evt_${Date.now()}`, name: t.name, description: t.description, icon: t.icon, unlocked: true, unlockedAt: Date.now(), condition: () => false });
   732→   732→        updates.achievementQueue = q;
   733→   733→        setTimeout(() => get().checkAchievements(), 0);
   734→   734→      }
   735→   735→
   736→   736→      // Power-up timer
   737→   737→      if (s.activePowerUp) {
   738→   738→        if (s.powerUpTimer > 0) updates.powerUpTimer = s.powerUpTimer - 1;
   739→   739→        else { updates.activePowerUp = null; updates.powerUpTimer = 0; }
   740→   740→      } else if (s.totalClicks > 20 && Math.random() < 0.0008) {
   741→   741→        const t = POWER_UP_DEFS[Math.floor(Math.random() * POWER_UP_DEFS.length)];
   742→   742→        if (t.effect === 'instantBonus') {
   743→   743→          const b = Math.round((s.autoRate * 30 + s.clickPower * 5) * 10) / 10;
   744→   744→          updates.crystals = Math.round(((updates.crystals ?? s.crystals) + b) * 100) / 100;
   745→   745→          updates.totalEarned = Math.round(((updates.totalEarned ?? s.totalEarned) + b) * 100) / 100;
   746→   746→          updates.crystalPulse = 2;
   747→   747→        } else {
   748→   748→          updates.activePowerUp = { ...t, timer: t.duration };
   749→   749→          updates.powerUpTimer = t.duration;
   750→   750→        }
   751→   751→      }
   752→   752→
   753→   753→      // Notification timer
   754→   754→      if (s.currentNotification) {
   755→   755→        if (s.notificationTimer > 0) updates.notificationTimer = s.notificationTimer - 1;
   756→   756→        else { updates.currentNotification = null; updates.notificationTimer = 0; }
   757→   757→      } else if (s.achievementQueue.length > 0) {
   758→   758→        const [next, ...rest] = s.achievementQueue;
   759→   759→        updates.currentNotification = next;
   760→   760→        updates.notificationTimer = 30;
   761→   761→        updates.achievementQueue = rest;
   762→   762→      }
   763→   763→
   764→   764→      // Click speed
   765→   765→      const now = Date.now();
   766→   766→      const r = s.clickTimestamps.filter(t => now - t < 1000);
   767→   767→      updates.clickTimestamps = r;
   768→   768→      updates.clicksPerSecond = r.length;
   769→   769→
   770→   770→      // Screen shake / crystal pulse clear
   771→   771→      if (s.screenShake) setTimeout(() => set({ screenShake: false }), 150);
   772→   772→      if (s.crystalPulse > 0) setTimeout(() => set({ crystalPulse: 0 }), 200);
   773→   773→
   774→   774→      if (Object.keys(updates).length > 0) set(updates);
   775→   775→    },
   776→   776→
   777→   777→    checkMilestones: () => {

   779→   779→      const nm = s.milestones.map(m => (!m.celebrated && s.totalEarned >= m.value ? { ...m, celebrated: true } : m));
   780→   780→      const just = nm.find((m, i) => m.celebrated && !s.milestones[i].celebrated);
   781→   781→      if (just) {
   782→   782→        set({ milestones: nm, screenShake: true, crystalPulse: 5 });
   783→   783→        get().addFloatingText(window.innerWidth / 2, window.innerHeight / 2, 0, 'milestone');
   784→   784→      }
   785→   785→    },

   862→   862→              const saved = su.find(s => s.id === u.id);

   882→   882→        : 0;
   883→   883→      set({
   884→   884→        crystals: (data.crystals as number) ?? 0, totalClicks: (data.totalClicks as number) ?? 0,
   885→   885→        totalEarned: (data.totalEarned as number) ?? 0, ...stats,
   886→   886→        prestige: (data.prestige as number) ?? 0, prestigePoints: (data.prestigePoints as number) ?? 0,
   887→   887→        goldenClicks: (data.goldenClicks as number) ?? 0, totalCrits: (data.totalCrits as number) ?? 0,
   888→   888→        maxCombo: (data.maxCombo as number) ?? 0, totalEvents: (data.totalEvents as number) ?? 0,
   889→   889→        upgrades: flattenUpgrades(areas),
   890→   890→        achievements, areas,
   891→   891→        currentArea: (data.currentArea as string) || 'crystal_caverns',
   892→   892→        lastOnlineTime: Date.now(),
   893→   893→        offlineEarned, showOfflineBonus: offlineEarned > 0,
