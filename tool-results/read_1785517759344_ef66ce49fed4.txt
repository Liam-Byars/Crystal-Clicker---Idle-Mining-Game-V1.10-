    19→export interface Achievement {
    20→  id: string;
    21→  name: string;
    22→  description: string;
    23→  icon: string;
    24→  unlocked: boolean;

    28→
    29→export interface FloatingText {

    31→  value: number;
    32→  x: number;

    38→export interface PowerUp {
    39→  id: string;
    40→  name: string;
    41→  icon: string;
    42→  effect: 'doubleClick' | 'tripleAuto' | 'instantBonus';
    43→  duration: number;
    44→  timer: number;
    45→  value: number;
    46→}
    47→
    48→export interface GameEvent {
    49→  id: string;
    50→  name: string;
    51→  description: string;
    52→  icon: string;
    53→  effect: 'doubleAuto' | 'doubleClick' | 'tripleGolden';

    55→  timer: number;
    56→  value: number;

    59→export interface Milestone {
    60→  id: number;
    61→  value: number;
    62→  label: string;
    63→  icon: string;
    64→  celebrated: boolean;

   817→export const getTotalCostN = (u: Upgrade, n: number): number => {
   818→  let t = 0;
   819→  for (let i = 0; i < n; i++) {
   820→    const c = getCostForLevel(u, u.level + i);

   822→    t += c;

   824→  return t;

   858→function recalcStats(upgrades: Upgrade[]) {
   859→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   860→  let goldenChance = 0.03; let critChance = 0.05;
   861→  for (const u of upgrades) {
   862→    for (let i = 0; i < u.level; i++) {
   863→      switch (u.effect) {
   864→        case 'clickPower': clickPower += u.value; break;
   865→        case 'autoRate': autoRate += u.value; break;
   866→        case 'multiplier': multiplier += u.value; break;
   867→        case 'goldenChance': goldenChance += u.value; break;
   868→        case 'critChance': critChance += u.value; break;
   869→      }
   870→    }
   871→  }
   872→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   873→}
