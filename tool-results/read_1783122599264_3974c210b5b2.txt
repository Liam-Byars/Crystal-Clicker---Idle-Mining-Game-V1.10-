   585→   585→function recalcStats(areas: Area[]): { clickPower: number; autoRate: number; multiplier: number; goldenChance: number; critChance: number } {
   586→   586→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   587→   587→  let goldenChance = 0.03; let critChance = 0.05;
   588→   588→  for (const area of areas) {
   589→   589→    if (!area.unlocked) continue;
   590→   590→    for (const u of area.upgrades) {
   591→   591→      for (let i = 0; i < u.level; i++) {
   592→   592→        switch (u.effect) {
   593→   593→          case 'clickPower': clickPower += u.value; break;
   594→   594→          case 'autoRate': autoRate += u.value; break;
   595→   595→          case 'multiplier': multiplier += u.value; break;
   596→   596→          case 'goldenChance': goldenChance += u.value; break;
   597→   597→          case 'critChance': critChance += u.value; break;
   598→   598→        }
   599→   599→      }
   600→   600→    }
   601→   601→  }
   602→   602→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.8), critChance: Math.min(critChance, 0.95) };
   603→   603→}
