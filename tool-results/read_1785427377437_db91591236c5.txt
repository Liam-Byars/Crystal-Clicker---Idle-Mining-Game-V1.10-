   544→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   545→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;
   546→  let total = 0; let count = 0;
   547→  for (let i = 0; i < cap; i++) {
   548→    const c = getCostForLevel(u, u.level + i);
   549→    if (total + c > money) break;
   550→    total += c; count++;
   551→  }
   552→  return count;
   553→};

   584→function recalcStats(upgrades: Upgrade[]) {
   585→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   586→  let goldenChance = 0.03; let critChance = 0.05;
   587→  for (const u of upgrades) {
   588→    for (let i = 0; i < u.level; i++) {
   589→      switch (u.effect) {
   590→        case 'clickPower': clickPower += u.value; break;
   591→        case 'autoRate': autoRate += u.value; break;
   592→        case 'multiplier': multiplier += u.value; break;
   593→        case 'goldenChance': goldenChance += u.value; break;
   594→        case 'critChance': critChance += u.value; break;
   595→      }
   596→    }
   597→  }
   598→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };
   599→}

   630→    let newCombo = 1; let newMax = s.maxCombo;
   631→    if (now - s.lastClickTime < 500) newCombo = Math.min(s.combo + 1, 50);
   632→    if (newCombo > newMax) newMax = newCombo;
   633→
   634→    const comboMult = 1 + (newCombo - 1) * 0.1;
   635→    const prestMult = 1 + s.prestigePoints * 0.1;
   636→    const puMult = s.activePowerUp?.effect === 'doubleClick' ? s.activePowerUp.value : 1;
   637→    const evMult = s.activeEvent?.effect === 'doubleClick' ? s.activeEvent.value : 1;

   650→      sessionClicks: s.sessionClicks + 1, sessionEarned: Math.round((s.sessionEarned + rv) * 100) / 100,
   651→      combo: newCombo, comboTimer: 60, maxCombo: newMax, lastClickTime: now,

   654→      crystalPulse: isCrit ? 3 : newCombo >= 5 ? 2 : 1,
   655→      floatingTextId: s.floatingTextId + 1,
   656→      clickTimestamps: recent, clicksPerSecond: recent.length,

   787→  // Batched tick — all 100ms updates in a single setState for performance
   788→  tick: () => {
   789→    const s = get();
   790→    const updates: Partial<GameState> = {};
   791→
   792→    // Auto income
   793→    if (s.autoRate > 0) {
   794→      let rate = s.autoRate;
   795→      const puBonus = s.activePowerUp?.effect === 'tripleAuto' ? s.activePowerUp.value : 1;
   796→      const evBonus = s.activeEvent?.effect === 'doubleAuto' ? s.activeEvent.value : 1;
   797→      rate *= puBonus * evBonus;
   798→      const prestMult = 1 + s.prestigePoints * 0.1;
   799→      const income = rate * prestMult * 0.1;
   800→      updates.crystals = Math.round((s.crystals + income) * 100) / 100;
   801→      updates.totalEarned = Math.round((s.totalEarned + income) * 100) / 100;
   802→      updates.sessionEarned = Math.round((s.sessionEarned + income) * 100) / 100;
   803→    }
   804→
   805→    // Combo decay
   806→    if (s.comboTimer > 0) {
   807→      updates.comboTimer = s.comboTimer - 1;
   808→      if (s.comboTimer <= 1) updates.combo = 0;
   809→    }
   810→
   811→    // Golden crystal
   812→    if (s.goldenActive) {
   813→      if (s.goldenTimer > 0) updates.goldenTimer = s.goldenTimer - 1;
   814→      else { updates.goldenActive = false; updates.goldenTimer = 0; }
   815→    } else {
   816→      let chance = 0.001;
   817→      if (s.activeEvent?.effect === 'tripleGolden') chance *= s.activeEvent.value;
   818→      chance *= (s.goldenChance / 0.03);
   819→      if (Math.random() < chance) {
   820→        const bv = s.clickPower * s.multiplier * 10 * (1 + s.prestigePoints * 0.1);
   821→        updates.goldenActive = true;
   822→        updates.goldenTimer = 400;
   823→        updates.goldenClickValue = Math.round(bv * 10) / 10;
   824→      }
   825→    }
   826→
   827→    // Event timer
   828→    if (s.activeEvent) {
   829→      if (s.eventTimer > 0) updates.eventTimer = s.eventTimer - 1;
   830→      else { updates.activeEvent = null; updates.eventTimer = 0; }
   831→    } else if (s.totalClicks > 50 && Math.random() < 0.0003) {
   832→      const t = EVENT_DEFS[Math.floor(Math.random() * EVENT_DEFS.length)];
   833→      updates.activeEvent = { ...t, timer: t.duration };
   834→      updates.eventTimer = t.duration;
   835→      updates.totalEvents = s.totalEvents + 1;
   836→      const q = [...s.achievementQueue];
   837→      q.push({ id: `evt_${Date.now()}`, name: t.name, description: t.description, icon: t.icon, unlocked: true, unlockedAt: Date.now(), condition: () => false });
   838→      updates.achievementQueue = q;
   839→      // Schedule achievement check after this set
   840→      setTimeout(() => get().checkAchievements(), 0);
   841→    }
   842→
   843→    // Power-up timer
   844→    if (s.activePowerUp) {
   845→      if (s.powerUpTimer > 0) updates.powerUpTimer = s.powerUpTimer - 1;
   846→      else { updates.activePowerUp = null; updates.powerUpTimer = 0; }
   847→    } else if (s.totalClicks > 20 && Math.random() < 0.0008) {
   848→      const t = POWER_UP_DEFS[Math.floor(Math.random() * POWER_UP_DEFS.length)];
   849→      if (t.effect === 'instantBonus') {
   850→        const b = Math.round((s.autoRate * 30 + s.clickPower * 5) * 10) / 10;
   851→        updates.crystals = Math.round(((updates.crystals ?? s.crystals) + b) * 100) / 100;
   852→        updates.totalEarned = Math.round(((updates.totalEarned ?? s.totalEarned) + b) * 100) / 100;
   853→        updates.crystalPulse = 2;
   854→      } else {
   855→        updates.activePowerUp = { ...t, timer: t.duration };
   856→        updates.powerUpTimer = t.duration;
   857→      }
   858→    }
   859→
   860→    // Notification timer
   861→    if (s.currentNotification) {
   862→      if (s.notificationTimer > 0) updates.notificationTimer = s.notificationTimer - 1;
   863→      else { updates.currentNotification = null; updates.notificationTimer = 0; }
   864→    } else if (s.achievementQueue.length > 0) {
   865→      const [next, ...rest] = s.achievementQueue;
   866→      updates.currentNotification = next;
   867→      updates.notificationTimer = 30;
   868→      updates.achievementQueue = rest;
   869→    }
   870→
   871→    // Click speed
   872→    const now = Date.now();
   873→    const r = s.clickTimestamps.filter(t => now - t < 1000);

   875→    updates.clicksPerSecond = r.length;
