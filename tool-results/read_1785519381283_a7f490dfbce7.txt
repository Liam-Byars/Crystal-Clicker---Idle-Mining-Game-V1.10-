   821→export const getTotalCostN = (u: Upgrade, n: number): number => {
   822→  let t = 0;
   823→  for (let i = 0; i < n; i++) {
   824→    const c = getCostForLevel(u, u.level + i);

   826→    t += c;
   827→  }
   828→  return t;

   862→function recalcStats(upgrades: Upgrade[]) {
   863→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   864→  let goldenChance = 0.03; let critChance = 0.05;
   865→  for (const u of upgrades) {
   866→    for (let i = 0; i < u.level; i++) {
   867→      switch (u.effect) {
   868→        case 'clickPower': clickPower += u.value; break;
   869→        case 'autoRate': autoRate += u.value; break;
   870→        case 'multiplier': multiplier += u.value; break;
   871→        case 'goldenChance': goldenChance += u.value; break;
   872→        case 'critChance': critChance += u.value; break;
   873→      }
   874→    }
   875→  }
   876→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   877→}
