    24→// ====== Helpers ======
    25→function fmt(n: number): string {
    26→  if (n < 0) return '-' + fmt(-n);
    27→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    28→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    29→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    30→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    31→  return (n / 1e12).toFixed(2) + 'T';
    32→}
    33→
    34→function fmtTime(ms: number): string {
    35→  const s = Math.floor(ms / 1000);
    36→  const h = Math.floor(s / 3600);
    37→  const m = Math.floor((s % 3600) / 60);
    38→  const sec = s % 60;
    39→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    40→  return `${m}:${String(sec).padStart(2, '0')}`;
    41→}
    42→
    43→function fmtTimer(ticks: number): string {
    44→  const sec = Math.ceil(ticks / 10);
    45→  const m = Math.floor(sec / 60);
    46→  const s = sec % 60;
    47→  return `${m}:${String(s).padStart(2, '0')}`;
    48→}

    71→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
    72→  if (!soundOn || !audioCtx) return;
    73→  try {
    74→    const o = audioCtx.createOscillator();
    75→    const g = audioCtx.createGain();
    76→    o.type = type;
    77→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    78→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    79→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    80→    o.connect(g);
    81→    g.connect(audioCtx.destination);
    82→    o.start();
    83→    o.stop(audioCtx.currentTime + dur);
    84→  } catch { /* ignore audio errors */ }
    85→}
