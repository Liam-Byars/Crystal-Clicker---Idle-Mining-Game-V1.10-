   704→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   705→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;
   706→  let total = 0; let count = 0;
   707→  for (let i = 0; i < cap; i++) {
   708→    const c = getCostForLevel(u, u.level + i);
   709→    if (total + c > money) break;
   710→    total += c; count++;
   711→  }
   712→  return count;

   744→function recalcStats(upgrades: Upgrade[]) {
   745→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   746→  let goldenChance = 0.03; let critChance = 0.05;
   747→  for (const u of upgrades) {
   748→    for (let i = 0; i < u.level; i++) {
   749→      switch (u.effect) {
   750→        case 'clickPower': clickPower += u.value; break;
   751→        case 'autoRate': autoRate += u.value; break;
   752→        case 'multiplier': multiplier += u.value; break;
   753→        case 'goldenChance': goldenChance += u.value; break;
   754→        case 'critChance': critChance += u.value; break;
   755→      }
   756→    }
   757→  }
   758→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };
   759→}
