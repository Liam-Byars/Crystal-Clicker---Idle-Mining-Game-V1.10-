     1→import { create } from 'zustand';
     2→
     3→// ====== Interfaces ======
     4→export interface Upgrade {
     5→  id: string;
     6→  name: string;
     7→  description: string;
     8→  icon: string;
     9→  baseCost: number;
    10→  costMultiplier: number;
    11→  level: number;
    12→  effect: 'clickPower' | 'autoRate' | 'multiplier' | 'goldenChance' | 'critChance';
    13→  value: number;
    14→  maxLevel?: number;
    15→}
    16→
    17→export interface Area {
    18→  id: string;
    19→  name: string;
    20→  description: string;
    21→  icon: string;
    22→  unlockCost: number;
    23→  unlocked: boolean;
    24→  upgrades: Upgrade[];
    25→  bgTint: string;        // CSS gradient for crystal area theming
    26→  accentColor: string;   // Tailwind color name for UI accents
    27→  headerGradient: string; // Gradient for area header
    28→  crystalType: string;     // Crystal/mineral name, e.g. "Crystal", "Amethyst"
    29→  crystalEmoji: string;    // Emoji representing the crystal, e.g. "🔴"
    30→  crystalGradient: string; // Tailwind gradient classes for crystal button
    31→  crystalGlow: string;     // Tailwind shadow class for crystal glow
    32→}
    33→
    34→export interface Achievement {
    35→  id: string;
    36→  name: string;
    37→  description: string;
    38→  icon: string;
    39→  unlocked: boolean;
    40→  unlockedAt?: number;
    41→  condition: (state: GameState) => boolean;
    42→}
    43→
    44→export interface FloatingText {
    45→  id: number;
    46→  value: number;
    47→  x: number;
    48→  y: number;
    49→  type: 'normal' | 'golden' | 'combo' | 'crit' | 'powerup' | 'event' | 'offline' | 'milestone';
    50→}
    51→
    52→export interface PowerUp {
    53→  id: string;
    54→  name: string;
    55→  icon: string;
    56→  effect: 'doubleClick' | 'tripleAuto' | 'instantBonus';
    57→  duration: number;
    58→  timer: number;
    59→  value: number;
    60→}
    61→
    62→export interface GameEvent {
    63→  id: string;
    64→  name: string;
    65→  description: string;
    66→  icon: string;
    67→  effect: 'doubleAuto' | 'doubleClick' | 'tripleGolden';
    68→  duration: number;
    69→  timer: number;
    70→  value: number;
    71→}
    72→
    73→export interface Milestone {
    74→  id: number;
    75→  value: number;
    76→  label: string;
    77→  icon: string;
    78→  celebrated: boolean;
    79→}
    80→
    81→export interface StoreItem {
    82→  id: string;
    83→  name: string;
    84→  description: string;
    85→  icon: string;
    86→  gemCost: number;
    87→  type: 'booster' | 'gem_pack';
    88→  boosterEffect?: {
    89→    type: 'doubleClick' | 'tripleAuto' | 'goldenFrenzy' | 'instantBonus';
    90→    duration: number; // in ticks (100ms each), so 600 = 60 seconds
    91→    value: number;
    92→  };
    93→  gemPackAmount?: number;
    94→}
    95→
    96→export interface OwnedBooster {
    97→  id: string;
    98→  itemId: string;
    99→  quantity: number;
   100→}
   101→
   102→export interface PremiumBooster {
   103→  itemId: string;
   104→  name: string;
   105→  icon: string;
   106→  effect: 'doubleClick' | 'tripleAuto' | 'goldenFrenzy' | 'instantBonus';
   107→  duration: number;
   108→  timer: number;
   109→  value: number;
   110→}
   111→
   112→export type BuyQuantity = 1 | 10 | 'max';
   113→
   114→export interface GameState {
   115→  // Core resources
   116→  crystals: number;
   117→  totalClicks: number;
   118→  totalEarned: number;
   119→  clickPower: number;
   120→  multiplier: number;
   121→  autoRate: number;
   122→
   123→  // Prestige
   124→  prestige: number;
   125→  prestigePoints: number;
   126→
   127→  // Combo
   128→  combo: number;
   129→  comboTimer: number;
   130→  maxCombo: number;
   131→  lastClickTime: number;
   132→
   133→  // Click speed tracking
   134→  clickTimestamps: number[];
   135→  clicksPerSecond: number;
   136→
   137→  // Critical hits
   138→  critChance: number;
   139→  critMultiplier: number;
   140→  totalCrits: number;
   141→
   142→  // Golden Crystal
   143→  goldenClicks: number;
   144→  goldenChance: number;
   145→  goldenActive: boolean;
   146→  goldenTimer: number;
   147→  goldenClickValue: number;
   148→
   149→  // Power-ups
   150→  activePowerUp: PowerUp | null;
   151→  powerUpTimer: number;
   152→
   153→  // Game Events
   154→  activeEvent: GameEvent | null;
   155→  eventTimer: number;
   156→  totalEvents: number;
   157→
   158→  // Premium currency & boosters
   159→  gems: number;
   160→  ownedBoosters: OwnedBooster[];
   161→  activePremiumBooster: PremiumBooster | null;
   162→  premiumBoosterTimer: number;
   163→
   164→  // Buy quantity
   165→  buyQuantity: BuyQuantity;
   166→
   167→  // Session tracking
   168→  sessionStartTime: number;
   169→  sessionClicks: number;
   170→  sessionEarned: number;
   171→
   172→  // Milestones
   173→  milestones: Milestone[];
   174→
   175→  // Offline earnings
   176→  lastOnlineTime: number;
   177→  offlineEarned: number;
   178→  showOfflineBonus: boolean;
   179→
   180→  // Floating texts
   181→  floatingTexts: FloatingText[];
   182→  floatingTextId: number;
   183→
   184→  // Achievement notifications
   185→  achievementQueue: Achievement[];
   186→  currentNotification: Achievement | null;
   187→  notificationTimer: number;
   188→
   189→  // Click ripples
   190→  ripples: { id: number; x: number; y: number; type: FloatingText['type'] }[];
   191→  rippleId: number;
   192→
   193→  // Upgrades & Achievements (legacy flat list kept for compat)
   194→  upgrades: Upgrade[];
   195→  achievements: Achievement[];
   196→
   197→  // Area System
   198→  areas: Area[];
   199→  currentArea: string;
   200→
   201→  // UI state
   202→  screenShake: boolean;
   203→  crystalPulse: number;
   204→  activeTab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'world' | 'store';
   205→
   206→  // Actions
   207→  click: (x: number, y: number) => void;
   208→  clickGolden: (x: number, y: number) => void;
   209→  buyUpgrade: (upgradeId: string, areaId?: string) => boolean;
   210→  setBuyQuantity: (q: BuyQuantity) => void;
   211→  performPrestige: () => void;
   212→  updateCombo: () => void;
   213→  updateGolden: () => void;
   214→  updatePowerUp: () => void;
   215→  updateEvent: () => void;
   216→  updateNotification: () => void;
   217→  updateClickSpeed: () => void;
   218→  checkMilestones: () => void;
   219→  addFloatingText: (x: number, y: number, value: number, type?: FloatingText['type']) => void;
   220→  removeFloatingText: (id: number) => void;
   221→  setActiveTab: (tab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'world' | 'store') => void;
   222→  checkAchievements: () => void;
   223→  resetGame: () => void;
   224→  loadSave: (data: Record<string, unknown>) => void;
   225→  getSaveData: () => Record<string, unknown>;
   226→  claimOfflineEarnings: () => void;
   227→  dismissOfflineBonus: () => void;
   228→  unlockArea: (areaId: string) => boolean;
   229→  switchArea: (areaId: string) => void;
   230→  tick: () => void;
   231→  purchaseStoreItem: (itemId: string) => boolean;
   232→  useBooster: (itemId: string) => boolean;
   233→  grantGems: (amount: number) => void;
   234→}

   585→function recalcStats(areas: Area[]): { clickPower: number; autoRate: number; multiplier: number; goldenChance: number; critChance: number } {
   586→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   587→  let goldenChance = 0.03; let critChance = 0.05;
   588→  for (const area of areas) {
   589→    if (!area.unlocked) continue;
   590→    for (const u of area.upgrades) {
   591→      for (let i = 0; i < u.level; i++) {
   592→        switch (u.effect) {
   593→          case 'clickPower': clickPower += u.value; break;
   594→          case 'autoRate': autoRate += u.value; break;
   595→          case 'multiplier': multiplier += u.value; break;
   596→          case 'goldenChance': goldenChance += u.value; break;
   597→          case 'critChance': critChance += u.value; break;
   598→        }
   599→      }
   600→    }
   601→  }
   602→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.8), critChance: Math.min(critChance, 0.95) };
   603→}

   880→      }

   883→      if (s.comboTimer > 0) {
   884→        updates.comboTimer = s.comboTimer - 1;
   885→        if (s.comboTimer <= 1) updates.combo = 0;
   886→      }
   887→
   888→      // Golden crystal
   889→      if (s.goldenActive) {
   890→        if (s.goldenTimer > 0) updates.goldenTimer = s.goldenTimer - 1;
   891→        else { updates.goldenActive = false; updates.goldenTimer = 0; }
   892→      } else {
   893→        let chance = 0.001;
   894→        if (s.activeEvent?.effect === 'tripleGolden') chance *= s.activeEvent.value;
   895→        if (s.activePremiumBooster?.effect === 'goldenFrenzy') chance *= s.activePremiumBooster.value;
   896→        chance *= (s.goldenChance / 0.03);
   897→        if (Math.random() < chance) {
   898→          const bv = s.clickPower * s.multiplier * 10 * (1 + s.prestigePoints * 0.1);
   899→          updates.goldenActive = true;
   900→          updates.goldenTimer = 400;
   901→          updates.goldenClickValue = Math.round(bv * 10) / 10;
   902→        }
   903→      }
   904→
   905→      // Event timer
   906→      if (s.activeEvent) {
   907→        if (s.eventTimer > 0) updates.eventTimer = s.eventTimer - 1;
   908→        else { updates.activeEvent = null; updates.eventTimer = 0; }

   910→        const t = EVENT_DEFS[Math.floor(Math.random() * EVENT_DEFS.length)];
   911→        updates.activeEvent = { ...t, timer: t.duration };
   912→        updates.eventTimer = t.duration;

   920→      // Power-up timer

   922→        if (s.powerUpTimer > 0) updates.powerUpTimer = s.powerUpTimer - 1;
   923→        else { updates.activePowerUp = null; updates.powerUpTimer = 0; }

   927→          const b = Math.round((s.autoRate * 30 + s.clickPower * 5) * 10) / 10;

   930→          updates.crystalPulse = 2;
   931→        } else {
   932→          updates.activePowerUp = { ...t, timer: t.duration };
   933→          updates.powerUpTimer = t.duration;
   934→        }
   935→      }
   936→
   937→      // Notification timer
   938→      if (s.currentNotification) {
   939→        if (s.notificationTimer > 0) updates.notificationTimer = s.notificationTimer - 1;
   940→        else { updates.currentNotification = null; updates.notificationTimer = 0; }
