   869→function recalcStats(upgrades: Upgrade[]) {
   870→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   871→  let goldenChance = 0.03; let critChance = 0.05;
   872→  for (const u of upgrades) {
   873→    for (let i = 0; i < u.level; i++) {
   874→      switch (u.effect) {
   875→        case 'clickPower': clickPower += u.value; break;
   876→        case 'autoRate': autoRate += u.value; break;
   877→        case 'multiplier': multiplier += u.value; break;
   878→        case 'goldenChance': goldenChance += u.value; break;
   879→        case 'critChance': critChance += u.value; break;
   880→      }
   881→    }
   882→  }
   883→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   884→}
