    40→function formatNumber(n: number): string {
    41→  if (n >= 1e15) return (n / 1e15).toFixed(2) + 'Q';
    42→  if (n >= 1e12) return (n / 1e12).toFixed(2) + 'T';
    43→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B';
    44→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M';
    45→  if (n >= 1e3) return (n / 1e3).toFixed(1) + 'K';
    46→  if (n >= 100) return Math.floor(n).toString();
    47→  if (n >= 10) return n.toFixed(1);
    48→  return n.toFixed(2);
    49→}

    75→// ====== Particle Burst on Click ======
    76→function ClickParticles({ x, y, isGolden }: { x: number; y: number; isGolden: boolean }) {
    77→  const particles = Array.from({ length: isGolden ? 12 : 6 }, (_, i) => ({
    78→    id: i,
    79→    angle: (i * 360) / 12 + Math.random() * 30,
    80→    distance: 40 + Math.random() * 60,
    81→    size: 3 + Math.random() * 4,
    82→  }));
    83→
    84→  return (
    85→    <div className="pointer-events-none fixed z-40" style={{ left: x, top: y }}>
    86→      {particles.map((p) => (
    87→        <motion.div
    88→          key={p.id}
    89→          className={`absolute rounded-full ${
    90→            isGolden
    91→              ? 'bg-amber-400'
    92→              : 'bg-purple-400'
    93→          }`}
    94→          style={{ width: p.size, height: p.size }}
    95→          initial={{ opacity: 1, x: 0, y: 0, scale: 1 }}
    96→          animate={{
    97→            opacity: 0,
    98→            x: Math.cos((p.angle * Math.PI) / 180) * p.distance,
    99→            y: Math.sin((p.angle * Math.PI) / 180) * p.distance,
   100→            scale: 0,
   101→          }}
   102→          transition={{ duration: 0.6, ease: 'easeOut' }}
   103→        />
   104→      ))}
   105→    </div>
   106→  );
   107→}
