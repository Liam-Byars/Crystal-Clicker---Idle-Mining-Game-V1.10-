    26→function formatNumber(n: number): string {
    27→  if (n >= 1e15) return (n / 1e15).toFixed(2) + 'Q';
    28→  if (n >= 1e12) return (n / 1e12).toFixed(2) + 'T';
    29→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B';
    30→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M';
    31→  if (n >= 1e3) return (n / 1e3).toFixed(1) + 'K';
    32→  if (n >= 100) return Math.floor(n).toString();
    33→  if (n >= 10) return n.toFixed(1);
    34→  return n.toFixed(2);
    35→}

   131→// ====== Particle Burst on Click ======
   132→function ClickParticles({ x, y, type }: { x: number; y: number; type: FloatingText['type'] }) {
   133→  const count = type === 'crit' ? 16 : type === 'golden' ? 12 : type === 'combo' ? 8 : 6;
   134→  const colorClass = type === 'crit'
   135→    ? 'bg-red-400'
   136→    : type === 'golden'
   137→      ? 'bg-amber-400'
   138→      : type === 'combo'
   139→        ? 'bg-orange-400'
   140→        : 'bg-purple-400';
   141→
   142→  const particles = useMemo(() =>
   143→    Array.from({ length: count }, (_, i) => ({
   144→      id: i,
   145→      angle: (i * 360) / count + (Math.random() - 0.5) * 40,
   146→      distance: 30 + Math.random() * 80,
   147→      size: 2 + Math.random() * 5,
   148→    })),
   149→  [count],
   150→  );
   151→
   152→  return (
   153→    <div className="pointer-events-none fixed z-40" style={{ left: x, top: y }}>
   154→      {particles.map((p) => (
   155→        <motion.div
   156→          key={p.id}
   157→          className={`absolute rounded-full ${colorClass}`}
   158→          style={{ width: p.size, height: p.size }}
   159→          initial={{ opacity: 1, x: 0, y: 0, scale: 1 }}
   160→          animate={{
   161→            opacity: 0,
   162→            x: Math.cos((p.angle * Math.PI) / 180) * p.distance,
   163→            y: Math.sin((p.angle * Math.PI) / 180) * p.distance,
   164→            scale: 0,
   165→          }}
   166→          transition={{ duration: 0.7, ease: 'easeOut' }}
   167→        />
   168→      ))}
   169→    </div>
   170→  );
   171→}
