   537→   744→function recalcStats(upgrades: Upgrade[]) {
   538→   745→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   539→   746→  let goldenChance = 0.03; let critChance = 0.05;
   540→   747→  for (const u of upgrades) {
   541→   748→    for (let i = 0; i < u.level; i++) {
   542→   749→      switch (u.effect) {
   543→   750→        case 'clickPower': clickPower += u.value; break;
   544→   751→        case 'autoRate': autoRate += u.value; break;
   545→   752→        case 'multiplier': multiplier += u.value; break;
   546→   753→        case 'goldenChance': goldenChance += u.value; break;
   547→   754→        case 'critChance': critChance += u.value; break;
   548→   755→      }
   549→   756→    }
   550→   757→  }
   551→   758→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };
   552→   759→}
