    18→function fmt(n: number): string {
    19→  if (n >= 1e15) return (n / 1e15).toFixed(2) + 'Q';
    20→  if (n >= 1e12) return (n / 1e12).toFixed(2) + 'T';
    21→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B';
    22→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M';
    23→  if (n >= 1e3) return (n / 1e3).toFixed(1) + 'K';
    24→  if (n >= 100) return Math.floor(n).toString();
    25→  if (n >= 10) return n.toFixed(1);
    26→  return n.toFixed(2);
    27→}
    28→
    29→function fmtTime(ms: number): string {
    30→  const s = Math.floor(ms / 1000);
    31→  const m = Math.floor(s / 60);
    32→  const h = Math.floor(m / 60);
    33→  if (h > 0) return `${h}:${String(m % 60).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;
    34→  return `${m}:${String(s % 60).padStart(2, '0')}`;
    35→}
