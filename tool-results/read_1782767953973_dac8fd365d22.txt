    27→function formatNumber(n: number): string {
    28→  if (n >= 1e15) return (n / 1e15).toFixed(2) + 'Q';
    29→  if (n >= 1e12) return (n / 1e12).toFixed(2) + 'T';
    30→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B';
    31→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M';
    32→  if (n >= 1e3) return (n / 1e3).toFixed(1) + 'K';
    33→  if (n >= 100) return Math.floor(n).toString();
    34→  if (n >= 10) return n.toFixed(1);
    35→  return n.toFixed(2);
    36→}

   132→// ====== Particle Burst on Click ======
   133→function ClickParticles({ x, y, type }: { x: number; y: number; type: FloatingText['type'] }) {
   134→  const count = type === 'crit' ? 16 : type === 'golden' ? 12 : type === 'combo' ? 8 : 6;
   135→  const colorClass = type === 'crit'
   136→    ? 'bg-red-400'
   137→    : type === 'golden'
   138→      ? 'bg-amber-400'
   139→      : type === 'combo'
   140→        ? 'bg-orange-400'
   141→        : 'bg-purple-400';
   142→
   143→  const particles = useMemo(() =>
   144→    Array.from({ length: count }, (_, i) => ({
   145→      id: i,
   146→      angle: (i * 360) / count + (Math.random() - 0.5) * 40,
   147→      distance: 30 + Math.random() * 80,
   148→      size: 2 + Math.random() * 5,
   149→    })),
   150→  [count],
   151→  );
   152→
   153→  return (
   154→    <div className="pointer-events-none fixed z-40" style={{ left: x, top: y }}>
   155→      {particles.map((p) => (
   156→        <motion.div
   157→          key={p.id}
   158→          className={`absolute rounded-full ${colorClass}`}
   159→          style={{ width: p.size, height: p.size }}
   160→          initial={{ opacity: 1, x: 0, y: 0, scale: 1 }}
   161→          animate={{
   162→            opacity: 0,
   163→            x: Math.cos((p.angle * Math.PI) / 180) * p.distance,
   164→            y: Math.sin((p.angle * Math.PI) / 180) * p.distance,
   165→            scale: 0,
   166→          }}
   167→          transition={{ duration: 0.7, ease: 'easeOut' }}
   168→        />
   169→      ))}
   170→    </div>
   171→  );
   172→}
