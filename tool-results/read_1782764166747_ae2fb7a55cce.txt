   254→   254→  return achievements.map(a => ({
   255→   255→    ...a,
   256→   256→    unlocked: false,
   257→   257→    condition: (state: GameState) => {
   258→   258→      switch (a.id) {
   259→   259→        case 'first_click': return state.totalClicks >= 1;
   260→   260→        case 'clicks_100': return state.totalClicks >= 100;
   261→   261→        case 'clicks_1000': return state.totalClicks >= 1000;
   262→   262→        case 'clicks_10000': return state.totalClicks >= 10000;
   263→   263→        case 'clicks_50000': return state.totalClicks >= 50000;
   264→   264→        case 'crystals_100': return state.totalEarned >= 100;
   265→   265→        case 'crystals_1000': return state.totalEarned >= 1000;
   266→   266→        case 'crystals_10000': return state.totalEarned >= 10000;
   267→   267→        case 'crystals_100000': return state.totalEarned >= 100000;
   268→   268→        case 'crystals_1m': return state.totalEarned >= 1000000;
   269→   269→        case 'upgrade_first': return state.upgrades.reduce((s, u) => s + u.level, 0) >= 1;
   270→   270→        case 'upgrade_10': return state.upgrades.reduce((s, u) => s + u.level, 0) >= 10;
   271→   271→        case 'upgrade_50': return state.upgrades.reduce((s, u) => s + u.level, 0) >= 50;
   272→   272→        case 'combo_5': return state.maxCombo >= 5;

   303→   303→export const getMaxBuyCount = (upgrade: Upgrade, availableCrystals: number): number => {
   304→   304→  if (upgrade.maxLevel) {
   305→   305→    const remaining = upgrade.maxLevel - upgrade.level;
   306→   306→    if (remaining <= 0) return 0;
   307→   307→    let totalCost = 0;
   308→   308→    let count = 0;
   309→   309→    for (let i = 0; i < remaining; i++) {
   310→   310→      const cost = getUpgradeCostForLevel(upgrade, upgrade.level + i);
   311→   311→      if (totalCost + cost > availableCrystals) break;
   312→   312→      totalCost += cost;
   313→   313→      count++;
   314→   314→    }
   315→   315→    return count;
   316→   316→  }
   317→   317→  let totalCost = 0;
   318→   318→  let count = 0;
   319→   319→  for (let i = 0; i < 10000; i++) { // safety limit
   320→   320→    const cost = getUpgradeCostForLevel(upgrade, upgrade.level + i);
   321→   321→    if (totalCost + cost > availableCrystals) break;
   322→   322→    totalCost += cost;
   323→   323→    count++;
   324→   324→  }
   325→   325→  return count;
   326→   326→};

   328→   328→export const getTotalCostForN = (upgrade: Upgrade, count: number): number => {
   329→   329→  let total = 0;
   330→   330→  for (let i = 0; i < count; i++) {
   331→   331→    total += getUpgradeCostForLevel(upgrade, upgrade.level + i);
   332→   332→  }
   333→   333→  return total;
   334→   334→};
