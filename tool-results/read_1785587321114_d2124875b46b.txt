   848→export const getTotalCostN = (u: Upgrade, n: number): number => {
   849→  let t = 0;
   850→  for (let i = 0; i < n; i++) {
   851→    const c = getCostForLevel(u, u.level + i);
   852→    if (!isFinite(c) || !isFinite(t + c)) return Infinity;
   853→    t += c;
   854→  }
   855→  return t;
   856→};

   889→function recalcStats(upgrades: Upgrade[]) {
   890→  // Compute all stats in log10 space to handle values beyond JS MAX_VALUE
   891→  let clickPowerLog = 0; // log10(clickPower), starts at 1 → log(1) = 0
   892→  let autoRateLog = -Infinity; // log10(autoRate), starts at 0 → -Infinity
   893→  let multiplierLog = 0; // log10(multiplier), starts at 1 → log(1) = 0
   894→  let goldenChance = 0.03; let critChance = 0.05;
   895→
   896→  for (const u of upgrades) {
   897→    if (u.level <= 0) continue;
   898→    switch (u.effect) {
   899→      case 'clickPower': {
   900→        // Total contribution = u.value * u.level
   901→        // log(u.value * u.level) = log(u.value) + log(u.level)
   902→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   903→        clickPowerLog = logAdd(clickPowerLog, contrib);
   904→        break;
   905→      }
   906→      case 'autoRate': {
   907→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   908→        autoRateLog = logAdd(autoRateLog, contrib);
   909→        break;
   910→      }
   911→      case 'multiplier': {
   912→        // Base amplifier adds 0.2 per level (additive to multiplier value)
   913→        // Area resonances add huge values (also additive)
   914→        // Use logAdd to correctly sum them in log space
   915→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   916→        multiplierLog = logAdd(multiplierLog, contrib);
   917→        break;
   918→      }
   919→      case 'goldenChance': goldenChance += u.value * u.level; break;
   920→      case 'critChance': critChance += u.value * u.level; break;
   921→    }
   922→  }
   923→  return {
   924→    clickPowerLog,
   925→    autoRateLog,
   926→    multiplierLog,
   927→    goldenChance: Math.min(goldenChance, 0.75),
   928→    critChance: Math.min(critChance, 1.0),
   929→  };
   930→}
