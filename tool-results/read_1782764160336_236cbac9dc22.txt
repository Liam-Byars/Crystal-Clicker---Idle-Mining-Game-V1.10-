   253→function buildAchievementConditions(achievements: Omit<Achievement, 'unlocked' | 'unlockedAt' | 'condition'>[]): Achievement[] {
   254→  return achievements.map(a => ({
   255→    ...a,
   256→    unlocked: false,
   257→    condition: (state: GameState) => {
   258→      switch (a.id) {
   259→        case 'first_click': return state.totalClicks >= 1;
   260→        case 'clicks_100': return state.totalClicks >= 100;
   261→        case 'clicks_1000': return state.totalClicks >= 1000;
   262→        case 'clicks_10000': return state.totalClicks >= 10000;
   263→        case 'clicks_50000': return state.totalClicks >= 50000;
   264→        case 'crystals_100': return state.totalEarned >= 100;
   265→        case 'crystals_1000': return state.totalEarned >= 1000;
   266→        case 'crystals_10000': return state.totalEarned >= 10000;
   267→        case 'crystals_100000': return state.totalEarned >= 100000;
   268→        case 'crystals_1m': return state.totalEarned >= 1000000;
   269→        case 'upgrade_first': return state.upgrades.reduce((s, u) => s + u.level, 0) >= 1;
   270→        case 'upgrade_10': return state.upgrades.reduce((s, u) => s + u.level, 0) >= 10;
   271→        case 'upgrade_50': return state.upgrades.reduce((s, u) => s + u.level, 0) >= 50;
   272→        case 'combo_5': return state.maxCombo >= 5;
   273→        case 'combo_10': return state.maxCombo >= 10;
   274→        case 'combo_25': return state.maxCombo >= 25;
   275→        case 'combo_50': return state.maxCombo >= 50;
   276→        case 'crit_first': return state.totalCrits >= 1;
   277→        case 'crit_100': return state.totalCrits >= 100;
   278→        case 'golden_first': return state.goldenClicks >= 1;
   279→        case 'golden_10': return state.goldenClicks >= 10;
   280→        case 'golden_50': return state.goldenClicks >= 50;
   281→        case 'prestige_1': return state.prestige >= 1;
   282→        case 'prestige_5': return state.prestige >= 5;

   303→export const getMaxBuyCount = (upgrade: Upgrade, availableCrystals: number): number => {
   304→  if (upgrade.maxLevel) {
   305→    const remaining = upgrade.maxLevel - upgrade.level;
   306→    if (remaining <= 0) return 0;
   307→    let totalCost = 0;
   308→    let count = 0;
   309→    for (let i = 0; i < remaining; i++) {
   310→      const cost = getUpgradeCostForLevel(upgrade, upgrade.level + i);
   311→      if (totalCost + cost > availableCrystals) break;
   312→      totalCost += cost;
   313→      count++;
   314→    }
   315→    return count;
   316→  }
   317→  let totalCost = 0;
   318→  let count = 0;
   319→  for (let i = 0; i < 10000; i++) { // safety limit
   320→    const cost = getUpgradeCostForLevel(upgrade, upgrade.level + i);
   321→    if (totalCost + cost > availableCrystals) break;
   322→    totalCost += cost;
   323→    count++;
   324→  }
   325→  return count;
   326→};

   332→  }
   333→  return total;

   412→  click: (x: number, y: number) => {
   413→    const state = get();
   414→    const now = Date.now();
   415→
   416→    // Click speed tracking
   417→    const recentClicks = [...state.clickTimestamps, now].filter(t => now - t < 1000);
   418→
   419→    // Combo system
   420→    let newCombo = 1;
   421→    let newMaxCombo = state.maxCombo;
   422→    if (now - state.lastClickTime < 500) {
   423→      newCombo = Math.min(state.combo + 1, 50);
   424→    }
   425→    if (newCombo > newMaxCombo) newMaxCombo = newCombo;
   426→
   427→    // Calculate click value
   428→    const comboMultiplier = 1 + (newCombo - 1) * 0.1;
   429→    const prestigeMultiplier = 1 + state.prestigePoints * 0.1;
   430→    const powerUpMultiplier = state.activePowerUp?.effect === 'doubleClick' ? state.activePowerUp.value : 1;
   431→    const eventMultiplier = state.activeEvent?.effect === 'doubleClick' ? state.activeEvent.value : 1;
   432→    let value = state.clickPower * state.multiplier * comboMultiplier * prestigeMultiplier * powerUpMultiplier * eventMultiplier;
   433→
   434→    // Critical hit
   435→    let isCrit = false;
   436→    if (Math.random() < state.critChance) {
   437→      isCrit = true;
   438→      value *= state.critMultiplier;
   439→    }
   440→
   441→    const roundedValue = Math.round(value * 10) / 10;
   442→    const textType: FloatingText['type'] = isCrit ? 'crit' : newCombo > 1 ? 'combo' : 'normal';
   443→
   444→    // Add ripple
   445→    const newRippleId = state.rippleId + 1;
   446→
   447→    set({
   448→      crystals: Math.round((state.crystals + roundedValue) * 100) / 100,
   449→      totalClicks: state.totalClicks + 1,
   450→      totalEarned: Math.round((state.totalEarned + roundedValue) * 100) / 100,
   451→      sessionClicks: state.sessionClicks + 1,
   452→      sessionEarned: Math.round((state.sessionEarned + roundedValue) * 100) / 100,
   453→      combo: newCombo,
   454→      comboTimer: 60,
   455→      maxCombo: newMaxCombo,
   456→      lastClickTime: now,
   457→      totalCrits: isCrit ? state.totalCrits + 1 : state.totalCrits,
   458→      screenShake: isCrit || newCombo >= 10,
   459→      crystalPulse: isCrit ? 3 : newCombo >= 5 ? 2 : 1,
   460→      floatingTextId: state.floatingTextId + 1,
   461→      clickTimestamps: recentClicks,
   462→      clicksPerSecond: recentClicks.length,
   463→      ripples: [...state.ripples.slice(-5), { id: newRippleId, x, y, type: textType }],
   464→      rippleId: newRippleId,
   465→    });
   466→
   467→    get().addFloatingText(x, y, roundedValue, textType);
   468→    get().checkAchievements();
   469→    get().checkMilestones();
   470→  },
