    18→    18→// ====== Sound Engine (Web Audio API) ======
    19→    19→const audioCtx = typeof window !== 'undefined' ? new (window.AudioContext || (window as unknown as Record<string, unknown>).webkitAudioContext)() : null;
    20→    20→let soundEnabled = true;
    21→    21→
    22→    22→function playTone(freq: number, duration: number, type: OscillatorType = 'sine', vol = 0.1) {
    23→    23→  if (!soundEnabled || !audioCtx) return;
    24→    24→  const osc = audioCtx.createOscillator();
    25→    25→  const gain = audioCtx.createGain();
    26→    26→  osc.type = type;
    27→    27→  osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
    28→    28→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
    29→    29→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
    30→    30→  osc.connect(gain);
    31→    31→  gain.connect(audioCtx.destination);
    32→    32→  osc.start();
    33→    33→  osc.stop(audioCtx.currentTime + duration);
    34→    34→}

    65→    65→// ====== Formatters ======
    66→    66→function fmt(n: number): string {
    67→    67→  if (n >= 1e12) return (n / 1e12).toFixed(2) + 'T';
    68→    68→  if (n >= 1e9) return (n / 1e9).toFixed(2) + 'B';
    69→    69→  if (n >= 1e6) return (n / 1e6).toFixed(2) + 'M';
    70→    70→  if (n >= 1e3) return (n / 1e3).toFixed(2) + 'K';
    71→    71→  if (n >= 100) return Math.floor(n).toString();
    72→    72→  if (n >= 10) return n.toFixed(1);
    73→    73→  return n.toFixed(n % 1 === 0 ? 0 : 1);
    74→    74→}
    75→    75→
    76→    76→function fmtTime(seconds: number): string {
    77→    77→  const s = Math.ceil(seconds);
    78→    78→  const m = Math.floor(s / 60);
    79→    79→  const sec = s % 60;
    80→    80→  return m > 0 ? `${m}:${sec.toString().padStart(2, '0')}` : `${sec}s`;
    81→    81→}
    82→    82→
    83→    83→function fmtDuration(ms: number): string {
    84→    84→  const totalSec = Math.floor(ms / 1000);
    85→    85→  const h = Math.floor(totalSec / 3600);
    86→    86→  const m = Math.floor((totalSec % 3600) / 60);
    87→    87→  const s = totalSec % 60;
    88→    88→  if (h > 0) return `${h}h ${m}m ${s}s`;
    89→    89→  if (m > 0) return `${m}m ${s}s`;
    90→    90→  return `${s}s`;
    91→    91→}

   338→   338→  const buyQty = useGameStore(useShallow(s => s.buyQuantity));
   339→   339→  const buyUpgrade = useGameStore(useShallow(s => s.buyUpgrade));

   341→   341→  const maxed = upgrade.maxLevel != null && upgrade.level >= upgrade.maxLevel;
   342→   342→
   343→   343→  let displayCost: number;
   344→   344→  let buyCount: number;
   345→   345→  let canAfford: boolean;
   346→   346→
   347→   347→  if (maxed) {
   348→   348→    displayCost = 0;
   349→   349→    buyCount = 0;
   350→   350→    canAfford = false;
   351→   351→  } else if (buyQty === 1) {
   352→   352→    displayCost = getUpgradeCost(upgrade);
   353→   353→    buyCount = 1;
   354→   354→    canAfford = crystals >= displayCost;
   355→   355→  } else if (buyQty === 10) {
   356→   356→    buyCount = Math.min(10, upgrade.maxLevel ? upgrade.maxLevel - upgrade.level : 10);
   357→   357→    displayCost = buyCount > 0 ? getTotalCostN(upgrade, buyCount) : 0;
   358→   358→    canAfford = crystals >= displayCost && buyCount > 0;
   359→   359→  } else {
   360→   360→    buyCount = getMaxBuyCount(upgrade, crystals);
   361→   361→    displayCost = buyCount > 0 ? getTotalCostN(upgrade, buyCount) : 0;
   362→   362→    canAfford = buyCount > 0;
   363→   363→  }
   364→   364→
   365→   365→  const handleBuy = () => {
   366→   366→    if (buyQty === 1) {
   367→   367→      buyUpgrade(upgrade.id);
   368→   368→    } else {
   369→   369→      const n = buyQty === 'max' ? buyCount : Math.min(10, upgrade.maxLevel ? upgrade.maxLevel - upgrade.level : 10);
   370→   370→      for (let i = 0; i < n; i++) {
   371→   371→        if (!buyUpgrade(upgrade.id)) break;
   372→   372→      }
   373→   373→    }
   374→   374→  };
   375→   375→
   376→   376→  const nextCost = maxed ? 0 : getUpgradeCost(upgrade);
   377→   377→  const progressPct = upgrade.maxLevel ? (upgrade.level / upgrade.maxLevel) * 100 : undefined;
   378→   378→
   379→   379→  return (
   380→   380→    <TooltipProvider>
   381→   381→      <Tooltip>
   382→   382→        <TooltipTrigger asChild>
   383→   383→          <Card
   384→   384→            className={`group relative cursor-pointer border transition-all duration-200 upgrade-card-hover ${
   385→   385→              maxed
   386→   386→                ? 'border-emerald-500/20 bg-emerald-950/20 shimmer-bg'
   387→   387→                : canAfford
   388→   388→                  ? 'border-purple-500/30 bg-purple-950/30 hover:border-purple-400/50 hover:shadow-lg hover:shadow-purple-500/10 hover:bg-purple-950/40'
   389→   389→                  : 'border-gray-700/30 bg-gray-900/20 opacity-70'
   390→   390→            }`}
   391→   391→            onClick={handleBuy}
   392→   392→          >
   393→   393→            <CardContent className="flex items-center gap-3 p-3">
   394→   394→              <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-lg text-xl ${
   395→   395→                maxed ? 'bg-emerald-900/40' : 'bg-purple-900/40'
   396→   396→              }`}>
   397→   397→                {upgrade.icon}

   666→   666→// ====== Income Sparkline ======
   667→   667→function IncomeSparkline() {
   668→   668→  const [history, setHistory] = useState<number[]>(() => Array(60).fill(0));
   669→   669→  const prevCrystals = useRef(0);
   670→   670→
   671→   671→  useEffect(() => {
   672→   672→    const interval = setInterval(() => {
   673→   673→      const crystals = useGameStore.getState().crystals;
   674→   674→      const income = Math.max(0, crystals - prevCrystals.current);
   675→   675→      prevCrystals.current = crystals;
   676→   676→      setHistory(h => [...h.slice(1), income]);
   677→   677→    }, 1000);
   678→   678→    return () => clearInterval(interval);
   679→   679→  }, []);
   680→   680→
   681→   681→  const maxVal = Math.max(1, ...history);
   682→   682→  const width = 200;
   683→   683→  const height = 40;
   684→   684→  const points = history.map((v, i) => {
   685→   685→    const x = (i / (history.length - 1)) * width;
   686→   686→    const y = height - (v / maxVal) * (height - 4) - 2;
   687→   687→    return `${x},${y}`;
   688→   688→  }).join(' ');
   689→   689→  const areaPath = `M0,${height} L${points.split(' ').map((p, i) => {
   690→   690→    const [x, y] = p.split(',');
   691→   691→    return i === 0 ? `${x},${y}` : `L${x},${y}`;
   692→   692→  }).join(' ')} L${width},${height} Z`;
   693→   693→
   694→   694→  return (
   695→   695→    <div className="rounded-lg border border-gray-800/30 bg-gray-900/20 p-3">
   696→   696→      <div className="mb-1 flex items-center justify-between">
   697→   697→        <span className="text-xs text-gray-500">📈 Income/Sec (60s)</span>
   698→   698→        <span className="text-xs font-bold text-green-300">{fmt(history[history.length - 1])}/s</span>
   699→   699→      </div>
   700→   700→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-10" preserveAspectRatio="none">
   701→   701→        <defs>
   702→   702→          <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   703→   703→            <stop offset="0%" stopColor="rgb(74, 222, 128)" stopOpacity="0.3" />
   704→   704→            <stop offset="100%" stopColor="rgb(74, 222, 128)" stopOpacity="0" />
   705→   705→          </linearGradient>
   706→   706→        </defs>
   707→   707→        <path d={areaPath} fill="url(#sparkGrad)" />
   708→   708→        <polyline points={points} fill="none" stroke="rgb(74, 222, 128)" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   709→   709→      </svg>
   710→   710→    </div>
   711→   711→  );
   712→   712→}
