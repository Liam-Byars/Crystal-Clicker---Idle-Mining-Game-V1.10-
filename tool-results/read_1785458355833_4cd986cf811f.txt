    25→// ====== Helpers ======
    26→function fmt(n: number): string {
    27→  if (n < 0) return '-' + fmt(-n);
    28→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    29→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    30→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    31→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    32→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    33→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    34→  const exp = Math.floor(Math.log10(n));
    35→  const tier = Math.floor((exp - 15) / 3);
    36→  if (tier >= 0 && tier < 676) {
    37→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    38→    const second = String.fromCharCode(65 + (tier % 26));
    39→    const divisor = Math.pow(10, 15 + tier * 3);
    40→    return (n / divisor).toFixed(2) + first + second;
    41→  }
    42→  // Beyond ZZ
    43→  return n.toExponential(2);
    44→}

    46→function fmtTime(ms: number): string {
    47→  const s = Math.floor(ms / 1000);
    48→  const h = Math.floor(s / 3600);
    49→  const m = Math.floor((s % 3600) / 60);
    50→  const sec = s % 60;
    51→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    52→  return `${m}:${String(sec).padStart(2, '0')}`;
    53→}

    55→function fmtTimer(ticks: number): string {
    56→  const sec = Math.ceil(ticks / 10);
    57→  const m = Math.floor(sec / 60);
    58→  const s = sec % 60;
    59→  return `${m}:${String(s).padStart(2, '0')}`;
    60→}

    68→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
    69→  if (!soundOn || !audioCtx) return;
    70→  try {
    71→    const o = audioCtx.createOscillator();
    72→    const g = audioCtx.createGain();
    73→    o.type = type;
    74→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    75→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    76→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    77→    o.connect(g);
    78→    g.connect(audioCtx.destination);
    79→    o.start();
    80→    o.stop(audioCtx.currentTime + dur);
    81→  } catch { /* ignore audio errors */ }
    82→}
