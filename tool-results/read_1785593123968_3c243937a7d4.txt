   385→   385→/** Get theme key for a given mine index (0-299) */
   386→   386→function getThemeKey(index: number): ThemeKey {
   387→   387→  if (index < 37) return 'deepSpace';
   388→   388→  if (index < 77) return 'galacticCore';
   389→   389→  if (index < 127) return 'intergalactic';
   390→   390→  if (index < 177) return 'dimensional';
   391→   391→  if (index < 227) return 'cosmicHorror';
   392→   392→  if (index < 277) return 'abstract';
   393→   393→  return 'beyond';
   394→   394→}
   395→   395→
   396→   396→/** Get theme-local index (0-based within the theme) */
   397→   397→function getThemeLocalIndex(index: number): number {
   398→   398→  if (index < 37) return index;
   399→   399→  if (index < 77) return index - 37;
   400→   400→  if (index < 127) return index - 77;
   401→   401→  if (index < 177) return index - 127;
   402→   402→  if (index < 227) return index - 177;
   403→   403→  if (index < 277) return index - 227;
   404→   404→  return index - 277;
   405→   405→}

   407→   407→/** Generate unlockAt value with slightly non-linear scaling */
   408→   408→function generateUnlockAt(index: number): number {
   409→   409→  // Non-linear: use a slight curve so early mines are closer together
   410→   410→  // and later mines spread out more
   411→   411→  const t = index / (TOTAL_MINES - 1); // 0 to 1
   412→   412→  // Apply a slight power curve (cubic) for non-linearity
   413→   413→  const curved = Math.pow(t, 1.15);
   414→   414→  const exponent = EXPONENT_START + curved * EXPONENT_RANGE;
   415→   415→  const raw = Math.pow(10, Math.min(exponent, 300)); // Cap at 10^300 (JS safe)
   416→   416→  return raw;
   417→   417→}
   418→   418→
   419→   419→/** Format a large number for description text */
   420→   420→function formatNumber(n: number): string {
   421→   421→  const exp = Math.floor(Math.log10(n));
   422→   422→  const suffixes = ['', 'K', 'M', 'B', 'T'];
   423→   423→  const doubleLetters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
   424→   424→  if (exp < 3) return n.toFixed(0);
   425→   425→  if (exp < 15) {
   426→   426→    const si = Math.floor((exp - 3) / 3);
   427→   427→    return (n / Math.pow(1000, si + 1)).toFixed(1) + suffixes[si + 1];
   428→   428→  }
   429→   429→  const dlIdx = Math.floor((exp - 15) / 3);
   430→   430→  const first = Math.floor(dlIdx / 26);
   431→   431→  const second = dlIdx % 26;
   432→   432→  if (first >= 26) return 'ZZ+';
   433→   433→  const letter1 = doubleLetters[first];
   434→   434→  const letter2 = doubleLetters[second];
   435→   435→  const divisor = Math.pow(10, 15 + dlIdx * 3);
   436→   436→  return (n / divisor).toFixed(1) + letter1 + letter2;
   437→   437→}

   439→   439→/** Cycle through a themed array with wrapping */
   440→   440→function pickFrom<T>(arr: readonly T[], index: number): T {
   441→   441→  return arr[index % arr.length];
   442→   442→}

   444→   444→/** Generate a unique snake_case id from a name */
   445→   445→function nameToId(name: string, index: number): string {
   446→   446→  // Convert name to snake_case, ensure uniqueness with index suffix
   447→   447→  const base = name
   448→   448→    .toLowerCase()
   449→   449→    .replace(/[^a-z0-9]+/g, '_')
   450→   450→    .replace(/^_|_$/g, '')
   451→   451→    .substring(0, 40); // Limit length
   452→   452→  return `gen_${index}_${base}`;
   453→   453→}

   455→   455→/** Calculate upgrade values based on mine index */
   456→   456→function getUpgradeValues(index: number): {
   457→   457→  clickPower: number;
   458→   458→  autoRate: number;
   459→   459→  resonance: number;
   460→   460→  luckValue: number;
   461→   461→  ultimate: number;
   462→   462→} {
   463→   463→  // Values grow exponentially - roughly 10x every ~20 mines
   464→   464→  // Starting from the last hand-crafted mine values:
   465→   465→  // clickPower: 2e24, autoRate: 1e24, resonance: 5e13, ultimate: 1e14
   466→   466→  const growthFactor = Math.min(Math.pow(10, (index / TOTAL_MINES) * (Math.log10(1e2040 / 1e14))), 1e280);
   467→   467→  const baseClick = 2e24;
   468→   468→  const baseAuto = 1e24;
   469→   469→  const baseResonance = 5e13;
   470→   470→  const baseUltimate = 1e14;
   471→   471→
   472→   472→  return {
   473→   473→    clickPower: baseClick * growthFactor,
   474→   474→    autoRate: baseAuto * growthFactor,
   475→   475→    resonance: baseResonance * growthFactor,
   476→   476→    luckValue: Math.max(0.00005, 0.0003 * Math.pow(0.98, index)),
   477→   477→    ultimate: baseUltimate * growthFactor,
   478→   478→  };
   479→   479→}

   483→   483→function generateAreas(): Area[] {
   484→   484→  const areas: Area[] = [];
   485→   485→
   486→   486→  for (let i = 0; i < TOTAL_MINES; i++) {
   487→   487→    const themeKey = getThemeKey(i);
   488→   488→    const theme = THEMES[themeKey];
   489→   489→    const localIdx = getThemeLocalIndex(i);
   490→   490→    const unlockAt = generateUnlockAt(i);
   491→   491→    const palette = COLOR_PALETTES[i % COLOR_PALETTES.length];
   492→   492→
   493→   493→    const name = pickFrom(theme.names, localIdx);
   494→   494→    const id = nameToId(name, i);
   495→   495→
   496→   496→    areas.push({
   497→   497→      id,
   498→   498→      name,
   499→   499→      location: pickFrom(theme.locations, localIdx),
   500→   500→      flag: pickFrom(theme.icons, localIdx),
   501→   501→      gem: pickFrom(theme.gems, localIdx),
   502→   502→      description: pickFrom(theme.descs, localIdx),
   503→   503→      icon: pickFrom(theme.icons, localIdx),
   504→   504→      unlockAt,
   505→   505→      gradient: palette.gradient,
   506→   506→      glowColor: palette.glow,
   507→   507→      bgAccent: palette.bg,
   508→   508→    });
   509→   509→  }
   510→   510→
   511→   511→  return areas;
   512→   512→}
