   679→function recalcStats(upgrades: Upgrade[]) {
   680→  // Compute all stats in log10 space to handle values beyond JS MAX_VALUE
   681→  let clickPowerLog = 0; // log10(clickPower), starts at 1 → log(1) = 0
   682→  let autoRateLog = -Infinity; // log10(autoRate), starts at 0 → -Infinity
   683→  let multiplierLog = 0; // log10(multiplier), starts at 1 → log(1) = 0
   684→  let goldenChance = 0.03; let critChance = 0.05;
   685→
   686→  for (const u of upgrades) {
   687→    if (u.level <= 0) continue;
   688→    switch (u.effect) {
   689→      case 'clickPower': {
   690→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   691→        clickPowerLog = logAdd(clickPowerLog, contrib);
   692→        break;
   693→      }
   694→      case 'autoRate': {
   695→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   696→        autoRateLog = logAdd(autoRateLog, contrib);
   697→        break;
   698→      }
   699→      case 'multiplier': {
   700→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   701→        multiplierLog = logAdd(multiplierLog, contrib);
   702→        break;
   703→      }
   704→      case 'goldenChance': goldenChance += u.value * u.level; break;
   705→      case 'critChance': critChance += u.value * u.level; break;
   706→    }
   707→  }
   708→  return {
   709→    clickPowerLog,
   710→    autoRateLog,
   711→    multiplierLog,
   712→    goldenChance: Math.min(goldenChance, 0.75),
   713→    critChance: Math.min(critChance, 1.0),
   714→  };
   715→}
