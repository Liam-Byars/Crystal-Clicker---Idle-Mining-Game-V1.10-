    18→// ====== Helpers ======
    19→function fmt(n: number): string {
    20→  if (n < 0) return '-' + fmt(-n);
    21→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    22→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    23→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    24→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    25→  return (n / 1e12).toFixed(2) + 'T';
    26→}
    27→
    28→function fmtTime(ms: number): string {
    29→  const s = Math.floor(ms / 1000);
    30→  const h = Math.floor(s / 3600);
    31→  const m = Math.floor((s % 3600) / 60);
    32→  const sec = s % 60;
    33→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    34→  return `${m}:${String(sec).padStart(2, '0')}`;
    35→}
    36→
    37→function fmtTimer(ticks: number): string {
    38→  const sec = Math.ceil(ticks / 10);
    39→  const m = Math.floor(sec / 60);
    40→  const s = sec % 60;
    41→  return `${m}:${String(s).padStart(2, '0')}`;
    42→}

    50→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
    51→  if (!soundOn || !audioCtx) return;
    52→  try {
    53→    const o = audioCtx.createOscillator();
    54→    const g = audioCtx.createGain();
    55→    o.type = type;
    56→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    57→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    58→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    59→    o.connect(g);
    60→    g.connect(audioCtx.destination);
    61→    o.start();
    62→    o.stop(audioCtx.currentTime + dur);
    63→  } catch { /* ignore audio errors */ }
    64→}
