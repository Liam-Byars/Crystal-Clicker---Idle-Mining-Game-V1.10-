    28→// ====== Helpers ======
    29→function fmtExpLog(log: number): string {
    30→  if (log < 0) return '0';
    31→  if (log < 3) {
    32→    const v = Math.pow(10, log);
    33→    return v < 10 ? v.toFixed(1) : Math.floor(v).toString();
    34→  }
    35→  if (log < 6) return (Math.pow(10, log - 3)).toFixed(1) + 'K';
    36→  if (log < 9) return (Math.pow(10, log - 6)).toFixed(2) + 'M';
    37→  if (log < 12) return (Math.pow(10, log - 9)).toFixed(2) + 'B';
    38→  if (log < 15) return (Math.pow(10, log - 12)).toFixed(2) + 'T';
    39→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    40→  const tier = Math.floor((log - 15) / 3);
    41→  if (tier >= 0 && tier < 676) {
    42→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    43→    const second = String.fromCharCode(65 + (tier % 26));
    44→    const divisor_log = 15 + tier * 3;
    45→    return (Math.pow(10, log - divisor_log)).toFixed(2) + first + second;
    46→  }
    47→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ)
    48→  const tier2 = tier - 676;
    49→  if (tier2 >= 0 && tier2 < 17576) {
    50→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    51→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    52→    const c = String.fromCharCode(65 + (tier2 % 26));
    53→    const divisor_log = 15 + tier * 3;
    54→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c;
    55→  }
    56→  // Beyond ZZZ: quad-letter suffixes
    57→  const tier3 = tier2 - 17576;
    58→  if (tier3 >= 0 && tier3 < 456976) {
    59→    const a = String.fromCharCode(65 + Math.floor(tier3 / 17576));
    60→    const b = String.fromCharCode(65 + Math.floor((tier3 % 17576) / 676));
    61→    const c = String.fromCharCode(65 + Math.floor(((tier3 % 17576) % 676) / 26));
    62→    const d = String.fromCharCode(65 + (tier3 % 26));
    63→    const divisor_log = 15 + tier * 3;
    64→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c + d;
    65→  }
    66→  return Math.pow(10, log).toExponential(2);
    67→}

    69→function fmt(n: number, exp = 0): string {
    70→  if (n < 0) return '-' + fmt(-n, exp);
    71→  if (!isFinite(n)) return fmtExpLog(exp + 400);
    72→  if (exp > 0) return fmtExpLog(Math.log10(n) + exp);
    73→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    74→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    75→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    76→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    77→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    78→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    79→  const e = Math.floor(Math.log10(n));
    80→  const tier = Math.floor((e - 15) / 3);
    81→  if (tier >= 0 && tier < 676) {
    82→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    83→    const second = String.fromCharCode(65 + (tier % 26));
    84→    const divisor = Math.pow(10, 15 + tier * 3);
    85→    return (n / divisor).toFixed(2) + first + second;
    86→  }
    87→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ, 17,576 more tiers)
    88→  const tier2 = tier - 676;
    89→  if (tier2 >= 0 && tier2 < 17576) {
    90→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    91→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    92→    const c = String.fromCharCode(65 + (tier2 % 26));
    93→    const divisor = Math.pow(10, 15 + tier * 3);
    94→    return (n / divisor).toFixed(2) + a + b + c;
    95→  }
    96→  // Truly beyond all suffixes
    97→  return n.toExponential(2);
    98→}

   100→function fmtTime(ms: number): string {
   101→  const s = Math.floor(ms / 1000);
   102→  const h = Math.floor(s / 3600);
   103→  const m = Math.floor((s % 3600) / 60);
   104→  const sec = s % 60;
   105→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
   106→  return `${m}:${String(sec).padStart(2, '0')}`;
   107→}

   109→function fmtTimer(ticks: number): string {
   110→  const sec = Math.ceil(ticks / 10);
   111→  const m = Math.floor(sec / 60);
   112→  const s = sec % 60;
   113→  return `${m}:${String(s).padStart(2, '0')}`;
   114→}

   122→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
   123→  if (!soundOn || !audioCtx) return;
   124→  try {
   125→    const o = audioCtx.createOscillator();
   126→    const g = audioCtx.createGain();
   127→    o.type = type;
   128→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
   129→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
   130→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   131→    o.connect(g);
   132→    g.connect(audioCtx.destination);
   133→    o.start();
   134→    o.stop(audioCtx.currentTime + dur);
   135→  } catch { /* ignore audio errors */ }
   136→}
