   356→function buildAchievementConditions(): Achievement[] {
   357→  return ACHIEVEMENT_DEFS.map(a => ({
   358→    ...a,
   359→    unlocked: false,
   360→    condition: (s: GameState) => {
   361→      const ul = s.upgrades.reduce((sum, u) => sum + u.level, 0);
   362→      switch (a.id) {
   363→        case 'first_click': return s.totalClicks >= 1;
   364→        case 'clicks_100': return s.totalClicks >= 100;
   365→        case 'clicks_1000': return s.totalClicks >= 1000;
   366→        case 'clicks_10000': return s.totalClicks >= 10000;
   367→        case 'clicks_50000': return s.totalClicks >= 50000;
   368→        case 'crystals_100': return s.totalEarned >= 100;
   369→        case 'crystals_1000': return s.totalEarned >= 1000;
   370→        case 'crystals_10000': return s.totalEarned >= 10000;
   371→        case 'crystals_100000': return s.totalEarned >= 100000;
   372→        case 'crystals_1m': return s.totalEarned >= 1000000;
   373→        case 'upgrade_first': return ul >= 1;
   374→        case 'upgrade_10': return ul >= 10;
   375→        case 'upgrade_50': return ul >= 50;
   376→        case 'combo_5': return s.maxCombo >= 5;
   377→        case 'combo_10': return s.maxCombo >= 10;
   378→        case 'combo_25': return s.maxCombo >= 25;
   379→        case 'combo_50': return s.maxCombo >= 50;
   380→        case 'crit_first': return s.totalCrits >= 1;
   381→        case 'crit_100': return s.totalCrits >= 100;
   382→        case 'golden_first': return s.goldenClicks >= 1;
   383→        case 'golden_10': return s.goldenClicks >= 10;
   384→        case 'golden_50': return s.goldenClicks >= 50;
   385→        case 'prestige_1': return s.prestige >= 1;
   386→        case 'prestige_5': return s.prestige >= 5;
   387→        case 'auto_10': return s.autoRate >= 10;
   388→        case 'auto_100': return s.autoRate >= 100;
   389→        case 'speed_5': return s.clicksPerSecond >= 5;
   390→        case 'speed_10': return s.clicksPerSecond >= 10;

   404→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   405→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;
   406→  let total = 0; let count = 0;
   407→  for (let i = 0; i < cap; i++) {
   408→    const c = getCostForLevel(u, u.level + i);
   409→    if (total + c > money) break;
   410→    total += c; count++;
   411→  }
   412→  return count;
   413→};

   416→  let t = 0;
   417→  for (let i = 0; i < n; i++) t += getCostForLevel(u, u.level + i);
   418→  return t;

   444→function recalcStats(upgrades: Upgrade[]) {
   445→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   446→  let goldenChance = 0.03; let critChance = 0.05;
   447→  for (const u of upgrades) {
   448→    for (let i = 0; i < u.level; i++) {
   449→      switch (u.effect) {
   450→        case 'clickPower': clickPower += u.value; break;
   451→        case 'autoRate': autoRate += u.value; break;
   452→        case 'multiplier': multiplier += u.value; break;
   453→        case 'goldenChance': goldenChance += u.value; break;
   454→        case 'critChance': critChance += u.value; break;
   455→      }
   456→    }
   457→  }
   458→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };
   459→}
