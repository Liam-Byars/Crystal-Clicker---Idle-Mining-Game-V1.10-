    17→// ====== Sound Engine ======
    18→const audioCtx = typeof window !== 'undefined'
    19→  ? new (window.AudioContext || (window as unknown as Record<string, unknown>).webkitAudioContext)()
    20→  : null;
    21→let soundEnabled = true;
    22→
    23→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.1) {
    24→  if (!soundEnabled || !audioCtx) return;
    25→  const o = audioCtx.createOscillator();
    26→  const g = audioCtx.createGain();
    27→  o.type = type;
    28→  o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    29→  g.gain.setValueAtTime(vol, audioCtx.currentTime);
    30→  g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    31→  o.connect(g); g.connect(audioCtx.destination);
    32→  o.start(); o.stop(audioCtx.currentTime + dur);
    33→}

    44→// ====== Number Formatting ======
    45→function fmt(n: number): string {
    46→  if (n < 0) return '-' + fmt(-n);
    47→  if (n < 1000) return n % 1 === 0 ? String(n) : n.toFixed(1);
    48→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    49→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    50→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    51→  return (n / 1e12).toFixed(2) + 'T';
    52→}
    53→
    54→function fmtTime(s: number): string {
    55→  const h = Math.floor(s / 3600);
    56→  const m = Math.floor((s % 3600) / 60);
    57→  const sec = s % 60;
    58→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    59→  return `${m}:${String(sec).padStart(2, '0')}`;
    60→}

    85→// ====== Buy quantity helpers ======
    86→function getBuyCount(qty: BuyQuantity, upgrade: Upgrade, crystals: number): number {
    87→  if (qty === 1) return 1;
    88→  if (qty === 10) return Math.min(10, upgrade.maxLevel ? upgrade.maxLevel - upgrade.level : 10);
    89→  return getMaxBuyCount(upgrade, crystals);
    90→}
    91→
    92→function getBuyCost(qty: BuyQuantity, upgrade: Upgrade, crystals: number): number {
    93→  const n = getBuyCount(qty, upgrade, crystals);
    94→  if (n <= 0) return Infinity;
    95→  if (qty === 1) return getUpgradeCost(upgrade);
    96→  return getTotalCostN(upgrade, n);
    97→}

   102→  return 'xMax';
   103→}
