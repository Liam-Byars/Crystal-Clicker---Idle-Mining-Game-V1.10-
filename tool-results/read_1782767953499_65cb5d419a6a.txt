   295→function buildAchievementConditions(achievements: Omit<Achievement, 'unlocked' | 'unlockedAt' | 'condition'>[]): Achievement[] {
   296→  return achievements.map(a => ({
   297→    ...a,
   298→    unlocked: false,
   299→    condition: (state: GameState) => {
   300→      switch (a.id) {
   301→        case 'first_click': return state.totalClicks >= 1;
   302→        case 'clicks_100': return state.totalClicks >= 100;
   303→        case 'clicks_1000': return state.totalClicks >= 1000;
   304→        case 'clicks_10000': return state.totalClicks >= 10000;
   305→        case 'clicks_50000': return state.totalClicks >= 50000;

   307→        case 'crystals_1000': return state.totalEarned >= 1000;

   315→        case 'combo_10': return state.maxCombo >= 10;
   316→        case 'combo_25': return state.maxCombo >= 25;
   317→        case 'combo_50': return state.maxCombo >= 50;

   345→export const getMaxBuyCount = (upgrade: Upgrade, availableCrystals: number): number => {
   346→  if (upgrade.maxLevel) {
   347→    const remaining = upgrade.maxLevel - upgrade.level;
   348→    if (remaining <= 0) return 0;
   349→    let totalCost = 0;
   350→    let count = 0;
   351→    for (let i = 0; i < remaining; i++) {
   352→      const cost = getUpgradeCostForLevel(upgrade, upgrade.level + i);
   353→      if (totalCost + cost > availableCrystals) break;
   354→      totalCost += cost;
   355→      count++;
   356→    }
   357→    return count;
   358→  }
   359→  let totalCost = 0;
   360→  let count = 0;
   361→  for (let i = 0; i < 10000; i++) { // safety limit
   362→    const cost = getUpgradeCostForLevel(upgrade, upgrade.level + i);
   363→    if (totalCost + cost > availableCrystals) break;
   364→    totalCost += cost;
   365→    count++;
   366→  }
   367→  return count;
   368→};

   459→  click: (x: number, y: number) => {
   460→    const state = get();
   461→    const now = Date.now();
   462→
   463→    // Click speed tracking
   464→    const recentClicks = [...state.clickTimestamps, now].filter(t => now - t < 1000);
   465→
   466→    // Combo system
   467→    let newCombo = 1;
   468→    let newMaxCombo = state.maxCombo;
   469→    if (now - state.lastClickTime < 500) {
   470→      newCombo = Math.min(state.combo + 1, 50);
   471→    }
   472→    if (newCombo > newMaxCombo) newMaxCombo = newCombo;
   473→
   474→    // Calculate click value
   475→    const comboMultiplier = 1 + (newCombo - 1) * 0.1;
   476→    const prestigeMultiplier = 1 + state.prestigePoints * 0.1;
   477→    const powerUpMultiplier = state.activePowerUp?.effect === 'doubleClick' ? state.activePowerUp.value : 1;
   478→    const eventMultiplier = state.activeEvent?.effect === 'doubleClick' ? state.activeEvent.value : 1;
   479→    let value = state.clickPower * state.multiplier * comboMultiplier * prestigeMultiplier * powerUpMultiplier * eventMultiplier;
   480→
   481→    // Critical hit
   482→    let isCrit = false;
   483→    if (Math.random() < state.critChance) {
   484→      isCrit = true;
   485→      value *= state.critMultiplier;
