   532→   532→function recalcStats(areas: Area[]): { clickPower: number; autoRate: number; multiplier: number; goldenChance: number; critChance: number } {
   533→   533→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   534→   534→  let goldenChance = 0.03; let critChance = 0.05;
   535→   535→  for (const area of areas) {
   536→   536→    if (!area.unlocked) continue;
   537→   537→    for (const u of area.upgrades) {
   538→   538→      for (let i = 0; i < u.level; i++) {
   539→   539→        switch (u.effect) {
   540→   540→          case 'clickPower': clickPower += u.value; break;
   541→   541→          case 'autoRate': autoRate += u.value; break;
   542→   542→          case 'multiplier': multiplier += u.value; break;
   543→   543→          case 'goldenChance': goldenChance += u.value; break;
   544→   544→          case 'critChance': critChance += u.value; break;
   545→   545→        }
   546→   546→      }
   547→   547→    }
   548→   548→  }
   549→   549→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.8), critChance: Math.min(critChance, 0.95) };
   550→   550→}

   759→   759→          updates.goldenTimer = 400;

   762→   762→      }

   764→   764→      // Event timer
   765→   765→      if (s.activeEvent) {
   766→   766→        if (s.eventTimer > 0) updates.eventTimer = s.eventTimer - 1;
   767→   767→        else { updates.activeEvent = null; updates.eventTimer = 0; }
   768→   768→      } else if (s.totalClicks > 50 && Math.random() < 0.0003) {

   780→   780→      if (s.activePowerUp) {

   782→   782→        else { updates.activePowerUp = null; updates.powerUpTimer = 0; }

   788→   788→          updates.totalEarned = Math.round(((updates.totalEarned ?? s.totalEarned) + b) * 100) / 100;
   789→   789→          updates.crystalPulse = 2;
   790→   790→        } else {
   791→   791→          updates.activePowerUp = { ...t, timer: t.duration };
   792→   792→          updates.powerUpTimer = t.duration;
   793→   793→        }
   794→   794→      }
   795→   795→
   796→   796→      // Notification timer
   797→   797→      if (s.currentNotification) {

   799→   799→        else { updates.currentNotification = null; updates.notificationTimer = 0; }

   886→   886→
   887→   887→      let areas: Area[];

   889→   889→        // New format: areas with upgrade levels — migrate old IDs if present
