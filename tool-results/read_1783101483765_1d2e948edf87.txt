   500→   500→function recalcStats(areas: Area[]): { clickPower: number; autoRate: number; multiplier: number; goldenChance: number; critChance: number } {
   501→   501→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   502→   502→  let goldenChance = 0.03; let critChance = 0.05;
   503→   503→  for (const area of areas) {
   504→   504→    if (!area.unlocked) continue;
   505→   505→    for (const u of area.upgrades) {
   506→   506→      for (let i = 0; i < u.level; i++) {
   507→   507→        switch (u.effect) {
   508→   508→          case 'clickPower': clickPower += u.value; break;
   509→   509→          case 'autoRate': autoRate += u.value; break;
   510→   510→          case 'multiplier': multiplier += u.value; break;
   511→   511→          case 'goldenChance': goldenChance += u.value; break;
   512→   512→          case 'critChance': critChance += u.value; break;
   513→   513→        }
   514→   514→      }
   515→   515→    }
   516→   516→  }
   517→   517→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.8), critChance: Math.min(critChance, 0.95) };
   518→   518→}
