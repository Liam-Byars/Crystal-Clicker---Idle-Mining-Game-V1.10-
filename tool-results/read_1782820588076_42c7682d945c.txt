   267→   267→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   268→   268→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;
   269→   269→  let total = 0; let count = 0;
   270→   270→  for (let i = 0; i < cap; i++) {
   271→   271→    const c = getCostForLevel(u, u.level + i);
   272→   272→    if (total + c > money) break;
   273→   273→    total += c; count++;
   274→   274→  }
   275→   275→  return count;
   276→   276→};

   307→   307→function recalcStats(upgrades: Upgrade[]) {
   308→   308→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   309→   309→  let goldenChance = 0.03; let critChance = 0.05;
   310→   310→  for (const u of upgrades) {
   311→   311→    for (let i = 0; i < u.level; i++) {
   312→   312→      switch (u.effect) {
   313→   313→        case 'clickPower': clickPower += u.value; break;
   314→   314→        case 'autoRate': autoRate += u.value; break;
   315→   315→        case 'multiplier': multiplier += u.value; break;
   316→   316→        case 'goldenChance': goldenChance += u.value; break;
   317→   317→        case 'critChance': critChance += u.value; break;
   318→   318→      }
   319→   319→    }
   320→   320→  }
   321→   321→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };
   322→   322→}
