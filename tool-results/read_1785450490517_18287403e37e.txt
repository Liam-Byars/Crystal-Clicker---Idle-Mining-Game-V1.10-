     4→// ====== Interfaces ======
     5→export interface Upgrade {
     6→  id: string;
     7→  name: string;
     8→  description: string;
     9→  icon: string;
    10→  baseCost: number;
    11→  costMultiplier: number;
    12→  level: number;
    13→  effect: 'clickPower' | 'autoRate' | 'multiplier' | 'goldenChance' | 'critChance';
    14→  value: number;
    15→  maxLevel?: number;
    16→}
    17→
    18→export interface Achievement {
    19→  id: string;
    20→  name: string;
    21→  description: string;
    22→  icon: string;
    23→  unlocked: boolean;
    24→  unlockedAt?: number;
    25→  condition: (state: GameState) => boolean;
    26→}
    27→
    28→export interface FloatingText {
    29→  id: number;
    30→  value: number;
    31→  x: number;
    32→  y: number;
    33→  type: 'normal' | 'golden' | 'combo' | 'crit' | 'powerup' | 'event' | 'offline' | 'milestone';
    34→}
    35→
    36→export interface PowerUp {
    37→  id: string;
    38→  name: string;
    39→  icon: string;
    40→  effect: 'doubleClick' | 'tripleAuto' | 'instantBonus';
    41→  duration: number;
    42→  timer: number;
    43→  value: number;
    44→}
    45→
    46→export interface GameEvent {
    47→  id: string;
    48→  name: string;
    49→  description: string;
    50→  icon: string;
    51→  effect: 'doubleAuto' | 'doubleClick' | 'tripleGolden';
    52→  duration: number;
    53→  timer: number;
    54→  value: number;
    55→}
    56→
    57→export interface Milestone {
    58→  id: number;
    59→  value: number;
    60→  label: string;
    61→  icon: string;
    62→  celebrated: boolean;
    63→}
    64→
    65→export interface Area {
    66→  id: string;
    67→  name: string;
    68→  location: string;
    69→  flag: string;
    70→  gem: string;
    71→  description: string;
    72→  icon: string;
    73→  unlockAt: number;
    74→  gradient: string;
    75→  glowColor: string;
    76→  bgAccent: string;
    77→}
    78→
    79→export type BuyQuantity = 1 | 10 | 100 | 'max';
    80→
    81→export interface GameState {
    82→  // Core resources
    83→  crystals: number;
    84→  totalClicks: number;
    85→  totalEarned: number;
    86→  clickPower: number;
    87→  multiplier: number;
    88→  autoRate: number;
    89→
    90→  // Prestige
    91→  prestige: number;
    92→  prestigePoints: number;
    93→
    94→  // Combo
    95→  combo: number;
    96→  comboTimer: number;
    97→  maxCombo: number;
    98→  lastClickTime: number;
    99→
   100→  // Click speed tracking
   101→  clickTimestamps: number[];
   102→  clicksPerSecond: number;
   103→
   104→  // Critical hits
   105→  critChance: number;
   106→  critMultiplier: number;
   107→  totalCrits: number;
   108→
   109→  // Golden Crystal
   110→  goldenClicks: number;
   111→  goldenChance: number;
   112→  goldenActive: boolean;
   113→  goldenTimer: number;
   114→  goldenClickValue: number;
   115→
   116→  // Power-ups
   117→  activePowerUp: PowerUp | null;
   118→  powerUpTimer: number;
   119→
   120→  // Game Events
   121→  activeEvent: GameEvent | null;
   122→  eventTimer: number;
   123→  totalEvents: number;
   124→
   125→  // Buy quantity
   126→  buyQuantity: BuyQuantity;
   127→
   128→  // Session tracking
   129→  sessionStartTime: number;
   130→  sessionClicks: number;
   131→  sessionEarned: number;
   132→
   133→  // Milestones
   134→  milestones: Milestone[];
   135→
   136→  // Offline earnings
   137→  lastOnlineTime: number;
   138→  offlineEarned: number;
   139→  showOfflineBonus: boolean;
   140→
   141→  // Floating texts
   142→  floatingTexts: FloatingText[];
   143→  floatingTextId: number;
   144→
   145→  // Achievement notifications
   146→  achievementQueue: Achievement[];
   147→  currentNotification: Achievement | null;
   148→  notificationTimer: number;
   149→
   150→  // Click ripples
   151→  ripples: { id: number; x: number; y: number; type: FloatingText['type'] }[];
   152→  rippleId: number;
   153→
   154→  // Upgrades & Achievements
   155→  upgrades: Upgrade[];
   156→  achievements: Achievement[];
   157→
   158→  // Area system
   159→  currentArea: string;
   160→  unlockedAreas: string[];
   161→
   162→  // UI state
   163→  screenShake: boolean;
   164→  crystalPulse: number;
   165→  activeTab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'map';
   166→
   167→  // Actions
   168→  click: (x: number, y: number) => void;
   169→  clickGolden: (x: number, y: number) => void;
   170→  buyUpgrade: (upgradeId: string) => boolean;
   171→  setBuyQuantity: (q: BuyQuantity) => void;
   172→  performPrestige: () => void;
   173→  updateCombo: () => void;
   174→  updateGolden: () => void;
   175→  updatePowerUp: () => void;
   176→  updateEvent: () => void;
   177→  updateNotification: () => void;
   178→  updateClickSpeed: () => void;
   179→  checkMilestones: () => void;
   180→  addFloatingText: (x: number, y: number, value: number, type?: FloatingText['type']) => void;
   181→  removeFloatingText: (id: number) => void;
   182→  setActiveTab: (tab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'map') => void;
   183→  switchArea: (areaId: string) => void;
   184→  checkAreaUnlocks: () => void;
   185→  checkAchievements: () => void;
   186→  resetGame: () => void;
   187→  loadSave: (data: Record<string, unknown>) => void;
   188→  getSaveData: () => Record<string, unknown>;
   189→  claimOfflineEarnings: () => void;
   190→  dismissOfflineBonus: () => void;
   191→}

   794→  let total = 0; let count = 0;
   795→  for (let i = 0; i < cap; i++) {
   796→    const c = getCostForLevel(u, u.level + i);
   797→    if (total + c > money) break;
   798→    total += c; count++;
   799→  }
   800→  return count;
   801→};

   832→function recalcStats(upgrades: Upgrade[]) {
   833→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   834→  let goldenChance = 0.03; let critChance = 0.05;
   835→  for (const u of upgrades) {
   836→    for (let i = 0; i < u.level; i++) {
   837→      switch (u.effect) {
   838→        case 'clickPower': clickPower += u.value; break;
   839→        case 'autoRate': autoRate += u.value; break;
   840→        case 'multiplier': multiplier += u.value; break;
   841→        case 'goldenChance': goldenChance += u.value; break;
   842→        case 'critChance': critChance += u.value; break;
   843→      }
   844→    }
   845→  }
   846→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   847→}
