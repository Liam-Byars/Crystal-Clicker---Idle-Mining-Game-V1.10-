     6→export interface Upgrade {
     7→  id: string;
     8→  name: string;

    10→  icon: string;
    11→  baseCost: number;
    12→  costMultiplier: number;
    13→  level: number;
    14→  effect: 'clickPower' | 'autoRate' | 'multiplier' | 'goldenChance' | 'critChance';
    15→  value: number;
    16→  maxLevel?: number;

    18→
    19→export interface Achievement {
    20→  id: string;
    21→  name: string;
    22→  description: string;
    23→  icon: string;
    24→  unlocked: boolean;
    25→  unlockedAt?: number;
    26→  condition: (state: GameState) => boolean;
    27→}
    28→
    29→export interface FloatingText {
    30→  id: number;
    31→  value: number;
    32→  x: number;
    33→  y: number;
    34→  type: 'normal' | 'golden' | 'combo' | 'crit' | 'powerup' | 'event' | 'offline' | 'milestone';
    35→}
    36→
    37→export interface PowerUp {
    38→  id: string;
    39→  name: string;
    40→  icon: string;
    41→  effect: 'doubleClick' | 'tripleAuto' | 'instantBonus';
    42→  duration: number;
    43→  timer: number;
    44→  value: number;
    45→}
    46→
    47→export interface GameEvent {
    48→  id: string;
    49→  name: string;
    50→  description: string;
    51→  icon: string;
    52→  effect: 'doubleAuto' | 'doubleClick' | 'tripleGolden';
    53→  duration: number;
    54→  timer: number;
    55→  value: number;
    56→}
    57→
    58→export interface Milestone {
    59→  id: number;
    60→  value: number;
    61→  label: string;
    62→  icon: string;
    63→  celebrated: boolean;
    64→}
    65→
    66→export interface Area {
    67→  id: string;
    68→  name: string;
    69→  location: string;
    70→  flag: string;
    71→  gem: string;
    72→  description: string;
    73→  icon: string;
    74→  unlockAt: number;
    75→  gradient: string;
    76→  glowColor: string;
    77→  bgAccent: string;

    80→export type BuyQuantity = 1 | 10 | 100 | 'max';

   122→  powerUpTimer: number;

   126→  eventTimer: number;
   127→  totalEvents: number;
   128→
   129→  // Buy quantity
   130→  buyQuantity: BuyQuantity;
   131→
   132→  // Session tracking
   133→  sessionStartTime: number;
   134→  sessionClicks: number;
   135→  sessionEarned: number;
   136→
   137→  // Milestones

   140→  // Offline earnings

   142→  offlineEarned: number;
   143→  showOfflineBonus: boolean;
   144→
   145→  // Floating texts
   146→  floatingTexts: FloatingText[];
   147→  floatingTextId: number;
   148→
   149→  // Achievement notifications
   150→  achievementQueue: Achievement[];
   151→  currentNotification: Achievement | null;
   152→  notificationTimer: number;

   154→  // Click ripples

   170→
   171→  // Actions
   172→  click: (x: number, y: number) => void;
   173→  clickGolden: (x: number, y: number) => void;
   174→  buyUpgrade: (upgradeId: string) => boolean;
   175→  setBuyQuantity: (q: BuyQuantity) => void;
   176→  performPrestige: () => void;

   180→  updateEvent: () => void;
   181→  updateNotification: () => void;
   182→  updateClickSpeed: () => void;

   184→  addFloatingText: (x: number, y: number, value: number, type?: FloatingText['type']) => void;
   185→  removeFloatingText: (id: number) => void;
   186→  setActiveTab: (tab: 'upgrades' | 'achievements' | 'stats' | 'prestige' | 'map') => void;
   187→  switchArea: (areaId: string) => void;
   188→  checkAreaUnlocks: () => void;
   189→  checkAchievements: () => void;
   190→  resetGame: () => void;
   191→  loadSave: (data: Record<string, unknown>) => void;

   193→  claimOfflineEarnings: () => void;
   194→  dismissOfflineBonus: () => void;
   195→  claimReward: (amount: number, prestige?: number) => void;
   196→}

   813→export const getTotalCostN = (u: Upgrade, n: number): number => {
   814→  let t = 0;
   815→  for (let i = 0; i < n; i++) {
   816→    const c = getCostForLevel(u, u.level + i);

   818→    t += c;
   819→  }
   820→  return t;

   854→function recalcStats(upgrades: Upgrade[]) {
   855→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   856→  let goldenChance = 0.03; let critChance = 0.05;
   857→  for (const u of upgrades) {
   858→    for (let i = 0; i < u.level; i++) {
   859→      switch (u.effect) {
   860→        case 'clickPower': clickPower += u.value; break;
   861→        case 'autoRate': autoRate += u.value; break;
   862→        case 'multiplier': multiplier += u.value; break;
   863→        case 'goldenChance': goldenChance += u.value; break;
   864→        case 'critChance': critChance += u.value; break;
   865→      }
   866→    }
   867→  }
   868→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   869→}
