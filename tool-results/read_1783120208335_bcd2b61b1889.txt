     3→// ====== Interfaces ======
     4→export interface Upgrade {
     5→  id: string;
     6→  name: string;
     7→  description: string;
     8→  icon: string;
     9→  baseCost: number;
    10→  costMultiplier: number;
    11→  level: number;
    12→  effect: 'clickPower' | 'autoRate' | 'multiplier' | 'goldenChance' | 'critChance';
    13→  value: number;
    14→  maxLevel?: number;

    16→
    17→export interface Area {
    18→  id: string;
    19→  name: string;
    20→  description: string;
    21→  icon: string;
    22→  unlockCost: number;
    23→  unlocked: boolean;
    24→  upgrades: Upgrade[];

    26→  accentColor: string;   // Tailwind color name for UI accents
    27→  headerGradient: string; // Gradient for area header
    28→  crystalType: string;     // Crystal/mineral name, e.g. "Crystal", "Amethyst"

    30→  crystalGradient: string; // Tailwind gradient classes for crystal button
    31→  crystalGlow: string;     // Tailwind shadow class for crystal glow
    32→}
    33→
    34→export interface Achievement {
    35→  id: string;
    36→  name: string;
    37→  description: string;
    38→  icon: string;
    39→  unlocked: boolean;
    40→  unlockedAt?: number;
    41→  condition: (state: GameState) => boolean;
    42→}
    43→
    44→export interface FloatingText {
    45→  id: number;
    46→  value: number;
    47→  x: number;
    48→  y: number;
    49→  type: 'normal' | 'golden' | 'combo' | 'crit' | 'powerup' | 'event' | 'offline' | 'milestone';

    54→  name: string;

    58→  timer: number;
    59→  value: number;
    60→}
    61→

    63→  id: string;
    64→  name: string;
    65→  description: string;
    66→  icon: string;
    67→  effect: 'doubleAuto' | 'doubleClick' | 'tripleGolden';
    68→  duration: number;
    69→  timer: number;
    70→  value: number;

    72→
    73→export interface Milestone {
    74→  id: number;
    75→  value: number;
    76→  label: string;
    77→  icon: string;
    78→  celebrated: boolean;

   532→function recalcStats(areas: Area[]): { clickPower: number; autoRate: number; multiplier: number; goldenChance: number; critChance: number } {
   533→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   534→  let goldenChance = 0.03; let critChance = 0.05;
   535→  for (const area of areas) {
   536→    if (!area.unlocked) continue;
   537→    for (const u of area.upgrades) {
   538→      for (let i = 0; i < u.level; i++) {
   539→        switch (u.effect) {
   540→          case 'clickPower': clickPower += u.value; break;
   541→          case 'autoRate': autoRate += u.value; break;
   542→          case 'multiplier': multiplier += u.value; break;
   543→          case 'goldenChance': goldenChance += u.value; break;
   544→          case 'critChance': critChance += u.value; break;
   545→        }
   546→      }
   547→    }
   548→  }
   549→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.8), critChance: Math.min(critChance, 0.95) };
   550→}

   784→        const t = POWER_UP_DEFS[Math.floor(Math.random() * POWER_UP_DEFS.length)];
   785→        if (t.effect === 'instantBonus') {
   786→          const b = Math.round((s.autoRate * 30 + s.clickPower * 5) * 10) / 10;
   787→          updates.crystals = Math.round(((updates.crystals ?? s.crystals) + b) * 100) / 100;
   788→          updates.totalEarned = Math.round(((updates.totalEarned ?? s.totalEarned) + b) * 100) / 100;
   789→          updates.crystalPulse = 2;
   790→        } else {
   791→          updates.activePowerUp = { ...t, timer: t.duration };
   792→          updates.powerUpTimer = t.duration;
   793→        }
   794→      }
   795→
   796→      // Notification timer
   797→      if (s.currentNotification) {
   798→        if (s.notificationTimer > 0) updates.notificationTimer = s.notificationTimer - 1;
   799→        else { updates.currentNotification = null; updates.notificationTimer = 0; }
   800→      } else if (s.achievementQueue.length > 0) {
   801→        const [next, ...rest] = s.achievementQueue;
   802→        updates.currentNotification = next;
   803→        updates.notificationTimer = 30;
   804→        updates.achievementQueue = rest;
   805→      }
   806→
   807→      // Click speed
   808→      const now = Date.now();
   809→      const r = s.clickTimestamps.filter(t => now - t < 1000);
   810→      updates.clickTimestamps = r;
   811→      updates.clicksPerSecond = r.length;
   812→
   813→      // Screen shake / crystal pulse clear
   814→      if (s.screenShake) setTimeout(() => set({ screenShake: false }), 150);
   815→      if (s.crystalPulse > 0) setTimeout(() => set({ crystalPulse: 0 }), 200);
   816→
   817→      if (Object.keys(updates).length > 0) set(updates);
   818→    },
   819→
   820→    checkMilestones: () => {
   821→      const s = get();
   822→      const nm = s.milestones.map(m => (!m.celebrated && s.totalEarned >= m.value ? { ...m, celebrated: true } : m));
   823→      const just = nm.find((m, i) => m.celebrated && !s.milestones[i].celebrated);
   824→      if (just) {
   825→        set({ milestones: nm, screenShake: true, crystalPulse: 5 });
   826→        get().addFloatingText(window.innerWidth / 2, window.innerHeight / 2, 0, 'milestone');
   827→      }

   844→        return a;
   845→      });
   846→      set({ achievements: na, achievementQueue: q });

   873→      set({

   875→        totalEarned: Math.round((s.totalEarned + s.offlineEarned) * 100) / 100,
   876→        offlineEarned: 0, showOfflineBonus: false,

   881→    dismissOfflineBonus: () => set({ offlineEarned: 0, showOfflineBonus: false }),
   882→
   883→    loadSave: (data) => {
   884→      // Load areas data (new format)
   885→      const savedAreas = data.areas as { id: string; unlocked: boolean; upgradeLevels: Record<string, number> }[] | undefined;
   886→
   887→      let areas: Area[];
   888→      if (savedAreas && Array.isArray(savedAreas) && savedAreas.length > 0) {
   889→        // New format: areas with upgrade levels — migrate old IDs if present
   890→        areas = buildAreas().map(a => {
   891→          // Try to find saved area by new ID first, then by old migrated ID
   892→          let saved = savedAreas.find(sa => sa.id === a.id);
   893→          // If not found, check if an old area ID maps to this one
   894→          if (!saved) {
   895→            const oldId = Object.entries(AREA_ID_MIGRATION).find(([, newId]) => newId === a.id)?.[0];
   896→            if (oldId) saved = savedAreas.find(sa => sa.id === oldId);
   897→          }
   898→          if (!saved) return a;
   899→          const upgrades = a.upgrades.map(u => ({
   900→            ...u,
   901→            level: saved.upgradeLevels?.[u.id] ?? 0,
   902→          }));
   903→          return { ...a, unlocked: saved.unlocked, upgrades };
   904→        });
   905→      } else {
   906→        // Legacy format: flat upgrades array → migrate to Iron Range
   907→        const su = data.upgrades as { id: string; level: number }[] | undefined;
   908→        areas = buildAreas().map(a => {
   909→          if (a.id === 'iron_range' && su) {
   910→            const upgrades = a.upgrades.map(u => {
   911→              const saved = su.find(s => s.id === u.id);
   912→              return { ...u, level: saved?.level ?? 0 };
   913→            });
   914→            return { ...a, upgrades };
   915→          }
   916→          return a;
   917→        });
   918→      }
   919→
   920→      const stats = recalcStats(areas);
   921→      const la = data.achievements as { id: string; unlocked: boolean }[] | undefined;
   922→      let achievements = buildAchievementConditions();
   923→      if (la) achievements = achievements.map(a => {
   924→        const l = la.find(x => x.id === a.id);
   925→        return { ...a, unlocked: l?.unlocked ?? false };
   926→      });
   927→      const lastOnline = (data.lastOnlineTime as number) || Date.now();
   928→      const elapsed = Date.now() - lastOnline;
   929→      const offlineEarned = stats.autoRate > 0 && elapsed >= 60000
   930→        ? Math.round(stats.autoRate * (Math.min(elapsed, 8 * 3600000) / 1000) * 0.5 * 100) / 100
   931→        : 0;
   932→
   933→      // Migrate currentArea from old IDs
   934→      let currentArea = (data.currentArea as string) || 'iron_range';
   935→      currentArea = AREA_ID_MIGRATION[currentArea] || currentArea;
   936→      // Validate the migrated area exists
   937→      const validArea = areas.find(a => a.id === currentArea);
   938→      if (!validArea || !validArea.unlocked) currentArea = 'iron_range';
   939→
   940→      set({
   941→        crystals: (data.crystals as number) ?? 0, totalClicks: (data.totalClicks as number) ?? 0,
   942→        totalEarned: (data.totalEarned as number) ?? 0, ...stats,
   943→        prestige: (data.prestige as number) ?? 0, prestigePoints: (data.prestigePoints as number) ?? 0,
   944→        goldenClicks: (data.goldenClicks as number) ?? 0, totalCrits: (data.totalCrits as number) ?? 0,
   945→        maxCombo: (data.maxCombo as number) ?? 0, totalEvents: (data.totalEvents as number) ?? 0,
   946→        upgrades: flattenUpgrades(areas),
   947→        achievements, areas,
   948→        currentArea,
   949→        lastOnlineTime: Date.now(),
   950→        offlineEarned, showOfflineBonus: offlineEarned > 0,
   951→      });
   952→    },
   953→
   954→    getSaveData: () => {

   956→      return {
