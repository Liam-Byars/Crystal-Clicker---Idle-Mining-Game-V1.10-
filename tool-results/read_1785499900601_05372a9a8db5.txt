    25→// ====== Helpers ======
    26→function fmt(n: number): string {
    27→  if (n < 0) return '-' + fmt(-n);
    28→  if (!isFinite(n)) {
    29→    // Number overflowed JS max — show as ∞ZZ+
    30→    return '∞ZZ+';
    31→  }
    32→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    33→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    34→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    35→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    36→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    37→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    38→  const exp = Math.floor(Math.log10(n));
    39→  const tier = Math.floor((exp - 15) / 3);
    40→  if (tier >= 0 && tier < 676) {
    41→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    42→    const second = String.fromCharCode(65 + (tier % 26));
    43→    const divisor = Math.pow(10, 15 + tier * 3);
    44→    return (n / divisor).toFixed(2) + first + second;
    45→  }
    46→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ, 17,576 more tiers)
    47→  const tier2 = tier - 676;
    48→  if (tier2 >= 0 && tier2 < 17576) {
    49→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    50→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    51→    const c = String.fromCharCode(65 + (tier2 % 26));
    52→    const divisor = Math.pow(10, 15 + tier * 3);
    53→    return (n / divisor).toFixed(2) + a + b + c;
    54→  }
    55→  // Truly beyond all suffixes
    56→  return n.toExponential(2);
    57→}

    59→function fmtTime(ms: number): string {
    60→  const s = Math.floor(ms / 1000);
    61→  const h = Math.floor(s / 3600);
    62→  const m = Math.floor((s % 3600) / 60);
    63→  const sec = s % 60;
    64→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    65→  return `${m}:${String(sec).padStart(2, '0')}`;
    66→}

    68→function fmtTimer(ticks: number): string {
    69→  const sec = Math.ceil(ticks / 10);
    70→  const m = Math.floor(sec / 60);
    71→  const s = sec % 60;
    72→  return `${m}:${String(s).padStart(2, '0')}`;
    73→}

    81→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
    82→  if (!soundOn || !audioCtx) return;
    83→  try {
    84→    const o = audioCtx.createOscillator();
    85→    const g = audioCtx.createGain();
    86→    o.type = type;
    87→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    88→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    89→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    90→    o.connect(g);
    91→    g.connect(audioCtx.destination);
    92→    o.start();
    93→    o.stop(audioCtx.currentTime + dur);
    94→  } catch { /* ignore audio errors */ }
    95→}
