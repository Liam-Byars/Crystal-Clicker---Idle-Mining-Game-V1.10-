   385→/** Get theme key for a given mine index (0-299) */
   386→function getThemeKey(index: number): ThemeKey {
   387→  if (index < 37) return 'deepSpace';
   388→  if (index < 77) return 'galacticCore';
   389→  if (index < 127) return 'intergalactic';
   390→  if (index < 177) return 'dimensional';
   391→  if (index < 227) return 'cosmicHorror';
   392→  if (index < 277) return 'abstract';
   393→  return 'beyond';
   394→}

   396→/** Get theme-local index (0-based within the theme) */
   397→function getThemeLocalIndex(index: number): number {
   398→  if (index < 37) return index;
   399→  if (index < 77) return index - 37;
   400→  if (index < 127) return index - 77;
   401→  if (index < 177) return index - 127;
   402→  if (index < 227) return index - 177;
   403→  if (index < 277) return index - 227;
   404→  return index - 277;
   405→}

   407→/** Generate unlockAt value with slightly non-linear scaling */
   408→function generateUnlockAt(index: number): number {
   409→  // Non-linear: use a slight curve so early mines are closer together
   410→  // and later mines spread out more
   411→  const t = index / (TOTAL_MINES - 1); // 0 to 1
   412→  // Apply a slight power curve (cubic) for non-linearity
   413→  const curved = Math.pow(t, 1.15);
   414→  const exponent = EXPONENT_START + curved * EXPONENT_RANGE;
   415→  const raw = Math.pow(10, Math.min(exponent, 300)); // Cap at 10^300 (JS safe)
   416→  return raw;
   417→}

   419→/** Format a large number for description text */
   420→function formatNumber(n: number): string {
   421→  const exp = Math.floor(Math.log10(n));
   422→  const suffixes = ['', 'K', 'M', 'B', 'T'];
   423→  const doubleLetters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
   424→  if (exp < 3) return n.toFixed(0);
   425→  if (exp < 12) {
   426→    const si = Math.floor((exp - 3) / 3);
   427→    return (n / Math.pow(1000, si + 1)).toFixed(1) + suffixes[si + 1];
   428→  }
   429→  const dlIdx = Math.floor((exp - 12) / 3);
   430→  const first = Math.floor(dlIdx / 26);
   431→  const second = dlIdx % 26;
   432→  if (first >= 26) return 'ZZ+';
   433→  const letter1 = doubleLetters[first];
   434→  const letter2 = doubleLetters[second];
   435→  return letter1 + letter2;
   436→}

   438→/** Cycle through a themed array with wrapping */
   439→function pickFrom<T>(arr: readonly T[], index: number): T {
   440→  return arr[index % arr.length];
   441→}

   443→/** Generate a unique snake_case id from a name */
   444→function nameToId(name: string, index: number): string {
   445→  // Convert name to snake_case, ensure uniqueness with index suffix
   446→  const base = name
   447→    .toLowerCase()
   448→    .replace(/[^a-z0-9]+/g, '_')
   449→    .replace(/^_|_$/g, '')
   450→    .substring(0, 40); // Limit length
   451→  return `gen_${index}_${base}`;
   452→}

   454→/** Calculate upgrade values based on mine index */
   455→function getUpgradeValues(index: number): {
   456→  clickPower: number;
   457→  autoRate: number;
   458→  resonance: number;
   459→  luckValue: number;
   460→  ultimate: number;
   461→} {
   462→  // Values grow exponentially - roughly 10x every ~20 mines
   463→  // Starting from the last hand-crafted mine values:
   464→  // clickPower: 2e24, autoRate: 1e24, resonance: 5e13, ultimate: 1e14
   465→  const growthFactor = Math.min(Math.pow(10, (index / TOTAL_MINES) * (Math.log10(1e2040 / 1e14))), 1e280);
   466→  const baseClick = 2e24;
   467→  const baseAuto = 1e24;
   468→  const baseResonance = 5e13;
   469→  const baseUltimate = 1e14;
   470→
   471→  return {
   472→    clickPower: baseClick * growthFactor,
   473→    autoRate: baseAuto * growthFactor,
   474→    resonance: baseResonance * growthFactor,
   475→    luckValue: Math.max(0.00005, 0.0003 * Math.pow(0.98, index)),
   476→    ultimate: baseUltimate * growthFactor,
   477→  };
   478→}

   482→function generateAreas(): Area[] {
   483→  const areas: Area[] = [];
   484→
   485→  for (let i = 0; i < TOTAL_MINES; i++) {
   486→    const themeKey = getThemeKey(i);
   487→    const theme = THEMES[themeKey];
   488→    const localIdx = getThemeLocalIndex(i);
   489→    const unlockAt = generateUnlockAt(i);
   490→    const palette = COLOR_PALETTES[i % COLOR_PALETTES.length];
   491→
   492→    const name = pickFrom(theme.names, localIdx);
   493→    const id = nameToId(name, i);
   494→
   495→    areas.push({
   496→      id,
   497→      name,
   498→      location: pickFrom(theme.locations, localIdx),
   499→      flag: pickFrom(theme.icons, localIdx),
   500→      gem: pickFrom(theme.gems, localIdx),
   501→      description: pickFrom(theme.descs, localIdx),
   502→      icon: pickFrom(theme.icons, localIdx),
   503→      unlockAt,
   504→      gradient: palette.gradient,
   505→      glowColor: palette.glow,
   506→      bgAccent: palette.bg,
   507→    });
   508→  }
   509→
   510→  return areas;
   511→}

   524→      ? `+${(values.luckValue * 100).toFixed(4)}% golden chance per level`
