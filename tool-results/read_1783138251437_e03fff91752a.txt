   307→   307→function recalcStats(upgrades: Upgrade[]) {
   308→   308→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   309→   309→  let goldenChance = 0.03; let critChance = 0.05;
   310→   310→  for (const u of upgrades) {
   311→   311→    for (let i = 0; i < u.level; i++) {
   312→   312→      switch (u.effect) {
   313→   313→        case 'clickPower': clickPower += u.value; break;
   314→   314→        case 'autoRate': autoRate += u.value; break;
   315→   315→        case 'multiplier': multiplier += u.value; break;
   316→   316→        case 'goldenChance': goldenChance += u.value; break;
   317→   317→        case 'critChance': critChance += u.value; break;
   318→   318→      }
   319→   319→    }
   320→   320→  }
   321→   321→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };

   501→   501→  tick: () => {
   502→   502→    const s = get();
   503→   503→    const updates: Partial<GameState> = {};
   504→   504→
   505→   505→    // Auto income
   506→   506→    if (s.autoRate > 0) {
   507→   507→      let rate = s.autoRate;
   508→   508→      const puBonus = s.activePowerUp?.effect === 'tripleAuto' ? s.activePowerUp.value : 1;
   509→   509→      const evBonus = s.activeEvent?.effect === 'doubleAuto' ? s.activeEvent.value : 1;
   510→   510→      rate *= puBonus * evBonus;
   511→   511→      const prestMult = 1 + s.prestigePoints * 0.1;
   512→   512→      const income = rate * prestMult * 0.1;
   513→   513→      updates.crystals = Math.round((s.crystals + income) * 100) / 100;
   514→   514→      updates.totalEarned = Math.round((s.totalEarned + income) * 100) / 100;
   515→   515→      updates.sessionEarned = Math.round((s.sessionEarned + income) * 100) / 100;
   516→   516→    }
   517→   517→
   518→   518→    // Combo decay
   519→   519→    if (s.comboTimer > 0) {
   520→   520→      updates.comboTimer = s.comboTimer - 1;
   521→   521→      if (s.comboTimer <= 1) updates.combo = 0;
   522→   522→    }
   523→   523→
   524→   524→    // Golden crystal
   525→   525→    if (s.goldenActive) {
   526→   526→      if (s.goldenTimer > 0) updates.goldenTimer = s.goldenTimer - 1;
   527→   527→      else { updates.goldenActive = false; updates.goldenTimer = 0; }
   528→   528→    } else {
   529→   529→      let chance = 0.001;
   530→   530→      if (s.activeEvent?.effect === 'tripleGolden') chance *= s.activeEvent.value;
   531→   531→      chance *= (s.goldenChance / 0.03);
   532→   532→      if (Math.random() < chance) {
   533→   533→        const bv = s.clickPower * s.multiplier * 10 * (1 + s.prestigePoints * 0.1);
   534→   534→        updates.goldenActive = true;
   535→   535→        updates.goldenTimer = 400;
   536→   536→        updates.goldenClickValue = Math.round(bv * 10) / 10;
   537→   537→      }
   538→   538→    }
   539→   539→
   540→   540→    // Event timer
   541→   541→    if (s.activeEvent) {
   542→   542→      if (s.eventTimer > 0) updates.eventTimer = s.eventTimer - 1;
   543→   543→      else { updates.activeEvent = null; updates.eventTimer = 0; }
   544→   544→    } else if (s.totalClicks > 50 && Math.random() < 0.0003) {
   545→   545→      const t = EVENT_DEFS[Math.floor(Math.random() * EVENT_DEFS.length)];
   546→   546→      updates.activeEvent = { ...t, timer: t.duration };
   547→   547→      updates.eventTimer = t.duration;
   548→   548→      updates.totalEvents = s.totalEvents + 1;
   549→   549→      const q = [...s.achievementQueue];
   550→   550→      q.push({ id: `evt_${Date.now()}`, name: t.name, description: t.description, icon: t.icon, unlocked: true, unlockedAt: Date.now(), condition: () => false });
   551→   551→      updates.achievementQueue = q;
   552→   552→      // Schedule achievement check after this set
   553→   553→      setTimeout(() => get().checkAchievements(), 0);
   554→   554→    }
   555→   555→
   556→   556→    // Power-up timer
   557→   557→    if (s.activePowerUp) {
   558→   558→      if (s.powerUpTimer > 0) updates.powerUpTimer = s.powerUpTimer - 1;
   559→   559→      else { updates.activePowerUp = null; updates.powerUpTimer = 0; }
   560→   560→    } else if (s.totalClicks > 20 && Math.random() < 0.0008) {
   561→   561→      const t = POWER_UP_DEFS[Math.floor(Math.random() * POWER_UP_DEFS.length)];
   562→   562→      if (t.effect === 'instantBonus') {
   563→   563→        const b = Math.round((s.autoRate * 30 + s.clickPower * 5) * 10) / 10;
   564→   564→        updates.crystals = Math.round(((updates.crystals ?? s.crystals) + b) * 100) / 100;
   565→   565→        updates.totalEarned = Math.round(((updates.totalEarned ?? s.totalEarned) + b) * 100) / 100;
   566→   566→        updates.crystalPulse = 2;
   567→   567→      } else {
   568→   568→        updates.activePowerUp = { ...t, timer: t.duration };
   569→   569→        updates.powerUpTimer = t.duration;
   570→   570→      }
   571→   571→    }
   572→   572→
   573→   573→    // Notification timer
   574→   574→    if (s.currentNotification) {
   575→   575→      if (s.notificationTimer > 0) updates.notificationTimer = s.notificationTimer - 1;
   576→   576→      else { updates.currentNotification = null; updates.notificationTimer = 0; }

   579→   579→      updates.currentNotification = next;
   580→   580→      updates.notificationTimer = 30;

   582→   582→    }
   583→   583→
   584→   584→    // Click speed
   585→   585→    const now = Date.now();
   586→   586→    const r = s.clickTimestamps.filter(t => now - t < 1000);

   588→   588→    updates.clicksPerSecond = r.length;
   589→   589→
   590→   590→    // Screen shake / crystal pulse clear
   591→   591→    if (s.screenShake) setTimeout(() => set({ screenShake: false }), 150);
   592→   592→    if (s.crystalPulse > 0) setTimeout(() => set({ crystalPulse: 0 }), 200);
