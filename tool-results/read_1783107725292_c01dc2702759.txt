    11→    11→function fmt(n: number): string {
    12→    12→  if (n < 0) return '-' + fmt(-n);
    13→    13→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    14→    14→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    15→    15→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    16→    16→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    17→    17→  return (n / 1e12).toFixed(2) + 'T';
    18→    18→}
