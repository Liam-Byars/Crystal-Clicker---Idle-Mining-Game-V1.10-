   897→   897→function recalcStats(upgrades: Upgrade[]) {
   898→   898→  // Compute all stats in log10 space to handle values beyond JS MAX_VALUE
   899→   899→  let clickPowerLog = 0; // log10(clickPower), starts at 1 → log(1) = 0
   900→   900→  let autoRateLog = -Infinity; // log10(autoRate), starts at 0 → -Infinity
   901→   901→  let multiplierLog = 0; // log10(multiplier), starts at 1 → log(1) = 0
   902→   902→  let goldenChance = 0.03; let critChance = 0.05;
   903→   903→
   904→   904→  for (const u of upgrades) {
   905→   905→    if (u.level <= 0) continue;
   906→   906→    switch (u.effect) {
   907→   907→      case 'clickPower': {
   908→   908→        // Total contribution = u.value * u.level
   909→   909→        // log(u.value * u.level) = log(u.value) + log(u.level)
   910→   910→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   911→   911→        clickPowerLog = logAdd(clickPowerLog, contrib);
   912→   912→        break;
   913→   913→      }
   914→   914→      case 'autoRate': {
   915→   915→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   916→   916→        autoRateLog = logAdd(autoRateLog, contrib);
   917→   917→        break;
   918→   918→      }
   919→   919→      case 'multiplier': {
   920→   920→        // Base amplifier adds 0.2 per level (additive to multiplier value)
   921→   921→        // Area resonances add huge values (also additive)
   922→   922→        // Use logAdd to correctly sum them in log space
   923→   923→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   924→   924→        multiplierLog = logAdd(multiplierLog, contrib);
   925→   925→        break;
   926→   926→      }
   927→   927→      case 'goldenChance': goldenChance += u.value * u.level; break;
   928→   928→      case 'critChance': critChance += u.value * u.level; break;
   929→   929→    }
   930→   930→  }
   931→   931→  return {
   932→   932→    clickPowerLog,
   933→   933→    autoRateLog,
   934→   934→    multiplierLog,
   935→   935→    goldenChance: Math.min(goldenChance, 0.75),
   936→   936→    critChance: Math.min(critChance, 1.0),
   937→   937→  };
   938→   938→}
