    23→// ====== Helpers ======
    24→function fmt(n: number): string {
    25→  if (n < 0) return '-' + fmt(-n);
    26→  if (n < 1000) return n < 10 ? n.toFixed(1) : Math.floor(n).toString();
    27→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    28→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    29→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    30→  return (n / 1e12).toFixed(2) + 'T';
    31→}
    32→
    33→function fmtTime(ms: number): string {
    34→  const s = Math.floor(ms / 1000);
    35→  const h = Math.floor(s / 3600);
    36→  const m = Math.floor((s % 3600) / 60);
    37→  const sec = s % 60;
    38→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
    39→  return `${m}:${String(sec).padStart(2, '0')}`;
    40→}
    41→
    42→function fmtTimer(ticks: number): string {
    43→  const sec = Math.ceil(ticks / 10);
    44→  const m = Math.floor(sec / 60);
    45→  const s = sec % 60;
    46→  return `${m}:${String(s).padStart(2, '0')}`;
    47→}

    55→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
    56→  if (!soundOn || !audioCtx) return;
    57→  try {
    58→    const o = audioCtx.createOscillator();
    59→    const g = audioCtx.createGain();
    60→    o.type = type;
    61→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
    62→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
    63→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
    64→    o.connect(g);
    65→    g.connect(audioCtx.destination);
    66→    o.start();
    67→    o.stop(audioCtx.currentTime + dur);
    68→  } catch { /* ignore audio errors */ }
    69→}
