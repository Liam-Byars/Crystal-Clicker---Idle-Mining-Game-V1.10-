    35→function toLogSafe(n: number): number {
    36→  if (!isFinite(n) || n <= 0) return -Infinity;
    37→  return Math.log10(n);
    38→}
    39→function fmtExpLog(log: number): string {
    40→  if (log < 0) return '0';
    41→  if (log < 3) {
    42→    const v = Math.pow(10, log);
    43→    return v < 10 ? v.toFixed(1) : Math.floor(v).toString();
    44→  }
    45→  if (log < 6) return (Math.pow(10, log - 3)).toFixed(1) + 'K';
    46→  if (log < 9) return (Math.pow(10, log - 6)).toFixed(2) + 'M';
    47→  if (log < 12) return (Math.pow(10, log - 9)).toFixed(2) + 'B';
    48→  if (log < 15) return (Math.pow(10, log - 12)).toFixed(2) + 'T';
    49→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    50→  const tier = Math.floor((log - 15) / 3);
    51→  if (tier >= 0 && tier < 676) {
    52→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    53→    const second = String.fromCharCode(65 + (tier % 26));
    54→    const divisor_log = 15 + tier * 3;
    55→    return (Math.pow(10, log - divisor_log)).toFixed(2) + first + second;
    56→  }
    57→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ)
    58→  const tier2 = tier - 676;
    59→  if (tier2 >= 0 && tier2 < 17576) {
    60→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
    61→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
    62→    const c = String.fromCharCode(65 + (tier2 % 26));
    63→    const divisor_log = 15 + tier * 3;
    64→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c;
    65→  }
    66→  // Beyond ZZZ: quad-letter suffixes
    67→  const tier3 = tier2 - 17576;
    68→  if (tier3 >= 0 && tier3 < 456976) {
    69→    const a = String.fromCharCode(65 + Math.floor(tier3 / 17576));
    70→    const b = String.fromCharCode(65 + Math.floor((tier3 % 17576) / 676));
    71→    const c = String.fromCharCode(65 + Math.floor(((tier3 % 17576) % 676) / 26));
    72→    const d = String.fromCharCode(65 + (tier3 % 26));
    73→    const divisor_log = 15 + tier * 3;
    74→    return (Math.pow(10, log - divisor_log)).toFixed(2) + a + b + c + d;
    75→  }
    76→  return Math.pow(10, log).toExponential(2);
    77→}

    79→function fmt(n: number, exp = 0): string {
    80→  if (n < 0) return '-' + fmt(-n, exp);
    81→  if (!isFinite(n)) return fmtExpLog(exp + 400);
    82→  if (exp > 0) return fmtExpLog(Math.log10(n) + exp);
    83→  if (n < 1000) { if (n === Math.floor(n)) return Math.floor(n).toString(); return n < 10 ? n.toFixed(1) : Math.floor(n).toString(); }
    84→  if (n < 1e6) return (n / 1e3).toFixed(1) + 'K';
    85→  if (n < 1e9) return (n / 1e6).toFixed(2) + 'M';
    86→  if (n < 1e12) return (n / 1e9).toFixed(2) + 'B';
    87→  if (n < 1e15) return (n / 1e12).toFixed(2) + 'T';
    88→  // AA-ZZ double-letter suffixes (676 tiers, each 1000x)
    89→  const e = Math.floor(Math.log10(n));
    90→  const tier = Math.floor((e - 15) / 3);
    91→  if (tier >= 0 && tier < 676) {
    92→    const first = String.fromCharCode(65 + Math.floor(tier / 26));
    93→    const second = String.fromCharCode(65 + (tier % 26));
    94→    const divisor = Math.pow(10, 15 + tier * 3);
    95→    return (n / divisor).toFixed(2) + first + second;
    96→  }
    97→  // Beyond ZZ: triple-letter suffixes (AAA-ZZZ, 17,576 more tiers)
    98→  const tier2 = tier - 676;
    99→  if (tier2 >= 0 && tier2 < 17576) {
   100→    const a = String.fromCharCode(65 + Math.floor(tier2 / 676));
   101→    const b = String.fromCharCode(65 + Math.floor((tier2 % 676) / 26));
   102→    const c = String.fromCharCode(65 + (tier2 % 26));
   103→    const divisor = Math.pow(10, 15 + tier * 3);
   104→    return (n / divisor).toFixed(2) + a + b + c;
   105→  }
   106→  // Truly beyond all suffixes
   107→  return n.toExponential(2);
   108→}

   110→function fmtTime(ms: number): string {
   111→  const s = Math.floor(ms / 1000);
   112→  const h = Math.floor(s / 3600);
   113→  const m = Math.floor((s % 3600) / 60);
   114→  const sec = s % 60;
   115→  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(sec).padStart(2, '0')}`;
   116→  return `${m}:${String(sec).padStart(2, '0')}`;
   117→}

   119→function fmtTimer(ticks: number): string {
   120→  const sec = Math.ceil(ticks / 10);
   121→  const m = Math.floor(sec / 60);
   122→  const s = sec % 60;
   123→  return `${m}:${String(s).padStart(2, '0')}`;
   124→}

   132→function playTone(freq: number, dur: number, type: OscillatorType = 'sine', vol = 0.08) {
   133→  if (!soundOn || !audioCtx) return;
   134→  try {
   135→    const o = audioCtx.createOscillator();
   136→    const g = audioCtx.createGain();
   137→    o.type = type;
   138→    o.frequency.setValueAtTime(freq, audioCtx.currentTime);
   139→    g.gain.setValueAtTime(vol, audioCtx.currentTime);
   140→    g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   141→    o.connect(g);
   142→    g.connect(audioCtx.destination);
   143→    o.start();
   144→    o.stop(audioCtx.currentTime + dur);
   145→  } catch { /* ignore audio errors */ }
   146→}
