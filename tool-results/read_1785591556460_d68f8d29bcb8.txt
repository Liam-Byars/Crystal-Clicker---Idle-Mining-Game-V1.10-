   854→export const getTotalCostN = (u: Upgrade, n: number): number => {
   855→  let t = 0;
   856→  for (let i = 0; i < n; i++) {
   857→    const c = getCostForLevel(u, u.level + i);
   858→    if (!isFinite(c) || !isFinite(t + c)) return Infinity;
   859→    t += c;
   860→  }
   861→  return t;
   862→};

   895→function recalcStats(upgrades: Upgrade[]) {
   896→  // Compute all stats in log10 space to handle values beyond JS MAX_VALUE
   897→  let clickPowerLog = 0; // log10(clickPower), starts at 1 → log(1) = 0
   898→  let autoRateLog = -Infinity; // log10(autoRate), starts at 0 → -Infinity
   899→  let multiplierLog = 0; // log10(multiplier), starts at 1 → log(1) = 0
   900→  let goldenChance = 0.03; let critChance = 0.05;
   901→
   902→  for (const u of upgrades) {
   903→    if (u.level <= 0) continue;
   904→    switch (u.effect) {
   905→      case 'clickPower': {
   906→        // Total contribution = u.value * u.level
   907→        // log(u.value * u.level) = log(u.value) + log(u.level)
   908→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   909→        clickPowerLog = logAdd(clickPowerLog, contrib);
   910→        break;
   911→      }
   912→      case 'autoRate': {
   913→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   914→        autoRateLog = logAdd(autoRateLog, contrib);
   915→        break;
   916→      }
   917→      case 'multiplier': {
   918→        // Base amplifier adds 0.2 per level (additive to multiplier value)
   919→        // Area resonances add huge values (also additive)
   920→        // Use logAdd to correctly sum them in log space
   921→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   922→        multiplierLog = logAdd(multiplierLog, contrib);
   923→        break;
   924→      }
   925→      case 'goldenChance': goldenChance += u.value * u.level; break;
   926→      case 'critChance': critChance += u.value * u.level; break;
   927→    }
   928→  }
   929→  return {
   930→    clickPowerLog,
   931→    autoRateLog,
   932→    multiplierLog,
   933→    goldenChance: Math.min(goldenChance, 0.75),
   934→    critChance: Math.min(critChance, 1.0),
   935→  };
   936→}
