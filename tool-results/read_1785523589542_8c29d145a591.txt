   872→function recalcStats(upgrades: Upgrade[]) {
   873→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   874→  let goldenChance = 0.03; let critChance = 0.05;
   875→  for (const u of upgrades) {
   876→    for (let i = 0; i < u.level; i++) {
   877→      switch (u.effect) {
   878→        case 'clickPower': clickPower += u.value; break;
   879→        case 'autoRate': autoRate += u.value; break;
   880→        case 'multiplier': multiplier += u.value; break;
   881→        case 'goldenChance': goldenChance += u.value; break;
   882→        case 'critChance': critChance += u.value; break;
   883→      }
   884→    }
   885→  }
   886→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   887→}
