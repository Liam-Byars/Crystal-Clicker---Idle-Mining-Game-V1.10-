   264→export const getUpgradeCost = (u: Upgrade) => Math.floor(u.baseCost * Math.pow(u.costMultiplier, u.level));
   265→export const getCostForLevel = (u: Upgrade, lvl: number) => Math.floor(u.baseCost * Math.pow(u.costMultiplier, lvl));

   267→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   268→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;
   269→  let total = 0; let count = 0;
   270→  for (let i = 0; i < cap; i++) {
   271→    const c = getCostForLevel(u, u.level + i);
   272→    if (total + c > money) break;
   273→    total += c; count++;
   274→  }
   275→  return count;
   276→};

   278→export const getTotalCostN = (u: Upgrade, n: number): number => {
   279→  let t = 0;
   280→  for (let i = 0; i < n; i++) t += getCostForLevel(u, u.level + i);
   281→  return t;

   307→function recalcStats(upgrades: Upgrade[]) {
   308→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   309→  let goldenChance = 0.03; let critChance = 0.05;
   310→  for (const u of upgrades) {
   311→    for (let i = 0; i < u.level; i++) {
   312→      switch (u.effect) {
   313→        case 'clickPower': clickPower += u.value; break;
   314→        case 'autoRate': autoRate += u.value; break;
   315→        case 'multiplier': multiplier += u.value; break;
   316→        case 'goldenChance': goldenChance += u.value; break;
   317→        case 'critChance': critChance += u.value; break;
   318→      }
   319→    }
   320→  }
   321→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.5), critChance: Math.min(critChance, 0.8) };
   322→}

   353→    const prestMult = 1 + s.prestigePoints * 0.1;
   354→    const puMult = s.activePowerUp?.effect === 'doubleClick' ? s.activePowerUp.value : 1;
   355→    const evMult = s.activeEvent?.effect === 'doubleClick' ? s.activeEvent.value : 1;
   356→    let value = s.clickPower * s.multiplier * comboMult * prestMult * puMult * evMult;
   357→
   358→    let isCrit = false;
   359→    if (Math.random() < s.critChance) { isCrit = true; value *= s.critMultiplier; }

   365→    set({
   366→      crystals: Math.round((s.crystals + rv) * 100) / 100,
   367→      totalClicks: s.totalClicks + 1, totalEarned: Math.round((s.totalEarned + rv) * 100) / 100,

   371→      screenShake: isCrit || newCombo >= 10,
   372→      crystalPulse: isCrit ? 3 : newCombo >= 5 ? 2 : 1,
   373→      floatingTextId: s.floatingTextId + 1,
   374→      clickTimestamps: recent, clicksPerSecond: recent.length,

   398→  buyUpgrade: (id) => {
   399→    const s = get(); const idx = s.upgrades.findIndex(u => u.id === id);
