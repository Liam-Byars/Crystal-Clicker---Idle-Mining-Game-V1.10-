   791→export const getMaxBuyCount = (u: Upgrade, money: number): number => {
   792→  const cap = u.maxLevel ? u.maxLevel - u.level : 10000;

   794→  for (let i = 0; i < cap; i++) {
   795→    const c = getCostForLevel(u, u.level + i);
   796→    if (total + c > money) break;
   797→    total += c; count++;
   798→  }
   799→  return count;

   831→function recalcStats(upgrades: Upgrade[]) {
   832→  let clickPower = 1; let autoRate = 0; let multiplier = 1;
   833→  let goldenChance = 0.03; let critChance = 0.05;
   834→  for (const u of upgrades) {
   835→    for (let i = 0; i < u.level; i++) {
   836→      switch (u.effect) {
   837→        case 'clickPower': clickPower += u.value; break;
   838→        case 'autoRate': autoRate += u.value; break;
   839→        case 'multiplier': multiplier += u.value; break;
   840→        case 'goldenChance': goldenChance += u.value; break;
   841→        case 'critChance': critChance += u.value; break;
   842→      }
   843→    }
   844→  }
   845→  return { clickPower, autoRate, multiplier, goldenChance: Math.min(goldenChance, 0.75), critChance: Math.min(critChance, 1.0) };
   846→}
