   840→  }

   875→function recalcStats(upgrades: Upgrade[]) {
   876→  // Compute all stats in log10 space to handle values beyond JS MAX_VALUE
   877→  let clickPowerLog = 0; // log10(clickPower), starts at 1 → log(1) = 0
   878→  let autoRateLog = -Infinity; // log10(autoRate), starts at 0 → -Infinity
   879→  let multiplierLog = 0; // log10(multiplier), starts at 1 → log(1) = 0
   880→  let goldenChance = 0.03; let critChance = 0.05;
   881→
   882→  for (const u of upgrades) {
   883→    if (u.level <= 0) continue;
   884→    switch (u.effect) {
   885→      case 'clickPower': {
   886→        // Total contribution = u.value * u.level
   887→        // log(u.value * u.level) = log(u.value) + log(u.level)
   888→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   889→        clickPowerLog = logAdd(clickPowerLog, contrib);
   890→        break;
   891→      }
   892→      case 'autoRate': {
   893→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   894→        autoRateLog = logAdd(autoRateLog, contrib);
   895→        break;
   896→      }
   897→      case 'multiplier': {
   898→        // Base amplifier adds 0.2 per level (additive to multiplier value)
   899→        // Area resonances add huge values (also additive)
   900→        // Use logAdd to correctly sum them in log space
   901→        const contrib = Math.log10(Math.max(u.value, 1e-300)) + Math.log10(u.level);
   902→        multiplierLog = logAdd(multiplierLog, contrib);
   903→        break;
   904→      }
   905→      case 'goldenChance': goldenChance += u.value * u.level; break;
   906→      case 'critChance': critChance += u.value * u.level; break;
   907→    }
   908→  }
   909→  return {
   910→    clickPowerLog,
   911→    autoRateLog,
   912→    multiplierLog,
   913→    goldenChance: Math.min(goldenChance, 0.75),
   914→    critChance: Math.min(critChance, 1.0),
   915→  };
   916→}
